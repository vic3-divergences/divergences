dvg_transylvania.1 = { #Hungarian friends starter
	title = dvg_transylvania.1.t
	desc = dvg_transylvania.1.d
	flavor = dvg_transylvania.1.f

	type = country_event
	placement = ROOT

	event_image = {
		video = "southamerica_election"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/southamerica/election"
	icon = "gfx/interface/icons/event_icons/waving_flag.dds" # "gfx/interface/icons/event_icons/waving_flag.dds"

	duration = 1

	trigger = {

	}
	immediate = {
		c:TRS = {
			save_scope_as = scope_TRS
		}
		c:BOH = {
			save_scope_as = scope_BOH
		}
		random_scope_character = {
			limit = {
				has_variable = is_duke_nitra
			}
			save_scope_as = nitra_scope
		}
	}

	option = {
		name = dvg_transylvania.1.a
		default_option = yes

		if = {
			limit = {
				c:HUN ?= this
			}
			trigger_event = {id = dvg_hungary.1}	
		}
		else_if = {
			limit = {
				c:TRS ?= this
			}
			trigger_event = {id = dvg_transylvania.2}	
		}
		else_if = {
			limit = {
				c:CRO ?= this
			}
			trigger_event = {id = dvg_croatia.1}	
		}
		else = {
			trigger_event = {id = dvg_banat.1}
		}
	}

}
