	duration = 1

	trigger = {

	}
	immediate = {
		c:WAL = {
			save_scope_as = scope_WAL
		}
		c:MOL = {
			save_scope_as = scope_MOL
		}
	}

	}

	option = {
		name = dvg_romania.1.a

			}

 		}
		else = {
			trigger_event = { id = dvg_romania.3 days = 7 }
		}	
	}

dvg_romania.4 = { #Dacia done
	title = dvg_romania.4.t
	desc = dvg_romania.4.d
	flavor = {
		triggered_flavor = {
			flavor = dvg_romania.4.f2
			trigger = {
				has_law = law_type:law_national_liberation
			}
		}
		triggered_flavor = {
			flavor = dvg_romania.4.f1
			trigger = {
				NOT = { has_law = law_type:law_national_liberation }
			}
		}
	}

	type = country_event
	placement = ROOT

	event_image = {
		video = "southamerica_election"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/southamerica/election"
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	duration = 1

	trigger = {

	}
	immediate = {

	}

	option = { #liberation finished
		name = dvg_romania.4.a
		default_option = yes

		trigger = {
			has_law = law_type:law_national_liberation
		}

		ig:ig_armed_forces = {
			set_interest_group_name = ig_armed_forces
		}
		ig:ig_petty_bourgeoisie= {
			set_interest_group_name = ig_petty_bourgeoisie
		}
		activate_law = law_type:law_census_voting
		add_loyalists = {
			value = 0.15
			pop_type = pop_type:soldiers
		}
		add_loyalists = {
			value = 0.15
			pop_type = pop_type:officers
		}
		
	}
	option = { #no liberation
		name = dvg_romania.4.b
		default_option = yes

		trigger = {
			NOT = {
				has_law = law_type:law_national_liberation
			}
		}

		
	}

}
