	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	title = reform_or_revolution.1.t
	desc = reform_or_revolution.1.d
	flavor = reform_or_revolution.1.f

	event_image = {
		video = "ip5_soviet_deliberation"
	}

	duration = 3

	cooldown = { days = very_long_modifier_time }

	trigger = {
	}

	immediate = {
		ig:ig_trade_unions ?= {
			save_scope_as = trade_unions_ig
		}
		if = {
			limit = {
				any_political_movement = {
					is_political_movement_type = movement_socialist
				}
			}
			random_political_movement = {
				limit = {
					is_political_movement_type = movement_socialist
				}
				save_scope_as = socialist_movement_scope
			}
		}
	}

	option = {
		name = reform_or_revolution.1.a
		ai_chance = {
			base = 10
			modifier = {
				trigger = {
					any_political_movement = {
						is_political_movement_type = movement_socialist
						political_movement_radicalism >= define:NPolitics|RADICALISM_PROTESTING_THRESHOLD
					}
				}
				add = 10
			}
			modifier = {
				trigger = {
					ig:ig_trade_unions ?= {
						leader ?= {
							has_socialist_ideology = yes
						}
					}
				}
				add = 10
			}
			modifier = {
				trigger = {
					has_factory_councils_law_or_amendment = yes
				}
				add = 50
			}
			modifier = {
				trigger = {
					ai_wants_radicals = yes
				}
				add = 10
			}
		}
		add_modifier = {
			name = modifier_heroic_era_revolution
			days = short_modifier_time
			is_decaying = yes
		}
		if = {
			limit = {
				NOT = {
					has_law_or_variant = law_type:law_council_republic
				}
			}
			add_radicals = {
				value = small_radicals
				interest_group = ig:ig_trade_unions
			}
		}
		else = {
			add_loyalists = {
				value = small_radicals
				interest_group = ig:ig_trade_unions
			}
		}
	}

	option = {
		name = reform_or_revolution.1.b
		ai_chance = {
			base = 10
			modifier = {
				trigger = {
					any_political_movement = {
						is_political_movement_type = movement_socialist
						political_movement_radicalism < define:NPolitics|RADICALISM_AGITATING_THRESHOLD
					}
				}
				add = 10
			}
			modifier = {
				trigger = {
					ig:ig_trade_unions ?= {
						leader ?= {
							NOT = {
								has_socialist_ideology = yes
							}
						}
					}
				}
				add = 10
			}
			modifier = {
				trigger = {
					NOT = { has_factory_councils_law_or_amendment = yes }
					any_active_party = {
						party_counts_as_socialist_party = yes
						party_has_members = yes
						any_member = {
							is_in_government = yes
						}
					}
				}
				add = 50
			}
			modifier = {
				trigger = {
					ai_avoids_radicals = yes
				}
				add = 10
			}
		}
		add_modifier = {
			name = modifier_heroic_era_reform
			days = short_modifier_time
			is_decaying = yes
		}
	}

	option = {
		name = reform_or_revolution.1.c
		default_option = yes
		ai_chance = {
			base = 5
			modifier = {
				trigger = {
					any_political_movement = {
						is_political_movement_type = movement_socialist
						political_movement_support < define:NPolitics|SUPPORT_RELEVANT_THRESHOLD
					}
				}
				add = 5
			}
			modifier = {
				trigger = {
					any_active_party = {
						party_counts_as_socialist_party = yes
						party_has_members = yes
						party_total_clout < 0.05
					}
				}
				add = 5
			}
			modifier = {
				trigger = {
					has_factory_councils_law_or_amendment = yes
				}
				add = -15
			}
			modifier = {
				trigger = {
					ai_avoids_radicals = yes
				}
				add = 10
			}
		}
		add_loyalists = {
			value = small_radicals
			strata = upper
		}
	}

	after = {
		je:je_reform_or_revolution ?= {
			if = {
				limit = {
					reform_or_revolution_variables_initialised = no

			}
		}
	}
}

namespace = factory_councils

factory_councils.1 = {
	type = country_event

	icon = "gfx/interface/icons/event_icons/event_revolutionary_flag.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	title = factory_councils.1.t
	desc = factory_councils.1.d
	flavor = factory_councils.1.f

	event_image = {
		video = "ip5_soviet_deliberation"
	}

	duration = 3

	cooldown = { days = long_modifier_time }

	category = revolution

	trigger = {
		country_at_risk_of_factory_councils = yes
		any_civil_war = {
			is_civil_war_type = revolution
			political_movement = {
				OR = {
					political_movement_support >= 0.10
					political_movement_radicalism >= 0.90
				}
				OR = {
					law_stance = {
						law = law_type:law_factory_councils
						value > neutral
					}
					ig:ig_trade_unions ?= {
						law_stance = {
							law = law_type:law_factory_councils
							value > neutral
						}
					}
				}
			}
			civil_war_progress_high = yes
		}
	}

	immediate = {
		ig:ig_trade_unions ?= {
			save_scope_as = trade_unions_ig
		}
		random_civil_war = {
			limit = {
				is_civil_war_type = revolution
				civil_war_progress_high = yes
			}
			civil_war_save_scopes = yes
		}
		ordered_scope_state = {
			limit = {
				is_incorporated = yes
				is_in_revolt = yes
			}
			order_by = {
				value = "ig_state_pol_strength_share(scope:trade_unions_ig)"
				multiply = this.gdp
			}
			position = 0
			save_scope_as = council_state_scope
		}
	}

	option = {
		name = factory_councils.1.a
		ai_chance = {
			base = 5
		}
		active_law:lawgroup_labour_associations ?= {
			add_amendment = {
				type = amendment_factory_councils
				sponsor = PREV.ig:ig_trade_unions
				cooldown = 6
			}
		}
		scope:council_state_scope ?= {
			add_radicals_in_state = {
				value = medium_radicals
			}
		}
		scope:ongoing_revolution ?= {
			add_civil_war_progress = revolution_advance_medium
		}
		if = {
			limit = {
				is_ai = yes
				authority >= define:NPolitics|IG_BOLSTER_COST
			}
			scope:ongoing_revolution_movement ?= {
				set_bolstering = yes
			}
		}
	}

	option = {
		name = factory_councils.1.b
		default_option = yes
		ai_chance = {
			base = 10
		}
		active_law:lawgroup_labour_associations ?= {
			add_amendment = {
				type = amendment_factory_councils
				sponsor = PREV.ig:ig_trade_unions
				cooldown = 6
			}
		}
		scope:council_state_scope ?= {
			add_radicals_in_state = {
				value = small_radicals
			}
		}
		if = {
			limit = {
				is_ai = yes
				authority >= define:NPolitics|IG_SUPPRESSION_COST
			}
			scope:ongoing_revolution_movement ?= {
				set_suppression = yes
			}
		}
	}

	option = {
		name = factory_councils.1.c
		trigger = {
			ig:ig_armed_forces ?= {
				is_insurrectionary = no
			}
			scope:ongoing_revolution_movement ?= {
				political_movement_military_support < 0.05
			}
			ROOT.army_power_projection > {
				value = ROOT.total_population
				divide = 1000
				multiply = scope:ongoing_revolution_movement.political_movement_support
			}
		}
		highlighted_option = yes
		show_as_unavailable = {
			always = yes
		}
		ai_chance = {
			base = 90
		}
		scope:ongoing_revolution ?= {
			add_civil_war_progress = revolution_retreat_very_large
		}
		add_radicals = {
			value = very_large_radicals
			pop_type = machinists
		}
		add_radicals = {
			value = large_radicals
			pop_type = laborers
		}
		if = {
			limit = {
				is_ai = yes
				authority >= define:NPolitics|IG_SUPPRESSION_COST
			}
			scope:ongoing_revolution_movement ?= {
				set_suppression = yes
			}
		}
	}
