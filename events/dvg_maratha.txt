dvg_maratha.2 = {
	type = country_event
	placement = ROOT

	title = dvg_maratha.2.t
	desc = dvg_maratha.2.d
	flavor = dvg_maratha.2.f

	event_image = {
		video = "gfx/event_pictures/southamerica_aristocrats.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/southamerica/aristocrats"

	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	duration = 3

	trigger = {
	}

	immediate = {
		random_country = {
			limit = {
				owner = {
					is_ai = yes
					relations:root >= relations_threshold:amicable
					country_has_primary_culture = cu:marathi
					is_subject = yes
					NOT = { THIS = ROOT }
					top_overlord = {
						THIS = root.top_overlord
					}
					this.gdp < root.gdp
				}
			}
			save_scope_as = marathi_union_scope
		}
		
	}


	
	option = {
		name = dvg_maratha.2.a
		default_option = yes
		hidden_effect = {
			every_state_in_canada = {
				limit = {
					has_claim_by = scope:canada_annex_scope
				}
				state_region = {
					remove_claim = scope:canada_annex_scope
					add_claim = ROOT
				}
			}
		}
		
		annex_with_incorporation = scope:marathi_union_scope
	}
	option = {


		name = dvg_maratha.2.b
		
		
		ai_chance = {
			base = 50
		}
	}
}

	on_opened_soundeffect = "event:/SFX/Events/southamerica/aristocrats"

	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	duration = 3

	trigger = {
		prestige > 70.0 
		has_technology_researched = romanticism
		is_at_war = no

			has_variable = recently_lost_war
			any_scope_state = {
				devastation >= 1 
			}
		}
		exists = c:BAL
		c:BAL = ROOT
	}

	#Bali Starting Event

	option = {
		name = dvg_maratha.3.a
		default_option = yes


		activate_law = law_type:law_national_supremacy


		add_loyalists = {
			value = 0.10
		}
		

		ai_chance = {
			base = 1
		}
	}
	option = {
		name = dvg_maratha.3.b
		default_option = yes

		activate_law = law_type:law_racial_segregation

		
		
		ai_chance = {

		}
	}
