			}

			add_journal_entry = {
				type = je_punjab_independence
			}
		}
		
		create_diplomatic_play = {
			target_country = c:PAN
			war = no
			type = dp_rebellion_annex
			
			add_war_goal = {
				holder = c:PAN
				type = independence
				target_country = c:AFG
			}
		}

			base = 99
		}
	}
