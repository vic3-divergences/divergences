dvg_burma.1 = {
	type = country_event
	placement = ROOT

	title = dvg_burma.1.t
	desc = dvg_burma.1.d
	flavor = dvg_burma.1.f

	event_image = {
		video = "gfx/event_pictures/asia_buddhism.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/southamerica/aristocrats"

	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	duration = 3

	trigger = {
	}

	#Burma Completion Event

	option = {
		name = dvg_burma.1.a
		default_option = yes


		add_loyalists = {
			value = 0.10
		}
		
		if = {
			limit = {
				country_average_cultural_acceptance = {
					target = cu:kachin
					value >= acceptance_status_3
				}
			}
			add_primary_culture = cu:kachin
		}

		if = {
			limit = {
				country_average_cultural_acceptance = {
					target = cu:shan
					value >= acceptance_status_3
				}
			}
			add_primary_culture = cu:shan
		}

		if = {
			limit = {
				country_average_cultural_acceptance = {
					target = cu:karen
					value >= acceptance_status_3
				}
			}
			add_primary_culture = cu:karen
		}

		if = {
			limit = {
				country_average_cultural_acceptance = {
					target = cu:lushai
					value >= acceptance_status_3
				}
			}
			add_primary_culture = cu:lushai
		}

		add_radicals = {
			value = 0.10
			interest_group = ig:ig_intelligentsia
		}

		ai_chance = {
			base = 1
		}
	}

	option = {
		name = dvg_burma.1.b
		default_option = yes


		add_loyalists = {
			value = 0.10
			interest_group = ig:ig_petty_bourgeoisie
		}

		add_modifier = {
			name = dvg_assimilate

		}
		
		every_scope_state = {
			limit = {
				is_isolated_from_market = no
				NOT = {
					state_region = {
						is_homeland = cu:burmese
					}
				}
			}
			state_region = {
				add_homeland = cu:burmese
			}
		}

		

		ai_chance = {
			base = 1
		}
	}
}

	duration = 3

	trigger = {
	}

	immediate = {
		if = {
			limit = {
				exists = c:BEI

					is_ai = yes
					is_subject = no
					has_variable = recently_lost_war
					has_truce_with = ROOT
				}
			}
			annex = c:BEI
		}
		if = {
			limit = {
				exists = c:IDN
				c:IDN = {
					is_ai = yes
					is_subject = no
					has_variable = recently_lost_war
					has_truce_with = ROOT
				}
			}
			annex = c:IDN
		}
		
	}


	
	option = {
		name = dvg_burma.2.a
		default_option = yes
		
		add_loyalists = {
			value = 0.55
		}

		random_scope_pop = {
			limit = {
				religion = rel:sunni
			}
			change_pop_religion = {
				target = rel:sunni
				value = 1
			}
		}

		ai_chance = {
			base = 50
		}
		if = {
			limit = {

			}
			change_pop_religion = {
				target = rel:sunni
				value = 1
			}
		}
		create_incident = {
			country = c:BAL
			value = 30
		}
		
		s:STATE_EAST_BORNEO = { 
			add_claim = ROOT

		s:STATE_WEST_BORNEO = { 
			add_claim = ROOT

			add_claim = ROOT 

		}

dvg_burma.3 = {

	event_image = {
		video = "gfx/event_pictures/asia_buddhism.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/southamerica/aristocrats"

	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	duration = 3

	trigger = {
		prestige > 70.0 
		has_technology_researched = romanticism
		is_at_war = no
		NOT = {
			has_variable = recently_lost_war
			any_scope_state = {
				devastation >= 1 
			}
		}
		exists = c:BAL
		c:BAL = ROOT
	}

	#Bali Starting Event

	option = {
		name = dvg_burma.3.a
		default_option = yes
		add_radicals = {
			value = 0.10
		}
		add_modifier = {
			name = dvg_tourism
			months = 30 
			is_decaying = yes
		}

		random_scope_state = {
			limit = {
				is_capital = yes
			}
			add_modifier = {
				name = dvg_tourism
				months = 30 
				is_decaying = yes
			}
		}

		activate_law = law_type:law_no_migration_controls
		
		ai_chance = {
			base = 1
		}
	}
	option = {
		name = dvg_burma.3.b
		default_option = yes
		add_loyalists = {
			value = 0.10
		}

		activate_law = law_type:law_migration_controls

		
		ai_chance = {
			base = 1
		}
	}
}
