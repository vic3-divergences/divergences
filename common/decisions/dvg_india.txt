
dvg_bharat_subcontinent = {
	is_shown = {
		any_primary_culture = {
			has_discrimination_trait_group = heritage_group_south_asian
		}

		any_primary_culture = {
			has_discrimination_trait = language_hindustani
		}
		NOT = {
			has_global_variable = dvg_bharata
		}
	}

    possible = {
		OR = {
			country_tier = empire
			country_tier = hegemony
		}
	}

    when_taken = {

		create_incident = {
			country = ROOT
			value = 20
		}
		
		every_state_region = {
			limit = {
				region = sr:region_madras
				region = sr:region_central_india
			}
			add_claim = ROOT
		}
		
		set_global_variable = dvg_bharata

}
	ai_chance = {
		value = 100
	}
}

dvg_sikh_embrace = {
	is_shown = {
	
		has_variable = dvg_sikh_accepted
		cu:panjabi = {
			root = {
				cultural_acceptance_base = { target = prev value >= acceptance_status_5 }
			}
		}
		NOT = {
		   has_variable = dvg_convert_sikh
		   exists = c:PAN
		}
	}

    possible = {
		
		NOT = {
			has_modifier = decision_cooldown
			ruler = {
				has_trait = bigoted
			}
		}
		any_interest_group = {
			leader = {
				religion = rel:sikh
			}
			is_powerful = yes
		}
		religion_percent_country = {
			target = rel:sikh
			value > 0.25
		}

		has_law = law_type:law_state_religion
	}

    when_taken = {

		trigger_event = {
			id = dvg_afghanistan.4
		}
		set_variable = dvg_convert_sikh
	}
	ai_chance = {
		value = 100
	}
}

dvg_tamil_embrace = {
	is_shown = {
	
		culture_percent_country = {
			target = cu:tamil
			value > 0.05
		}
		NOT = {
			country_has_primary_culture = cu:tamil
		}
		OR = {
			is_in_geographic_region = geographic_region_india
			is_in_geographic_region = geographic_region_southeast_asia
			is_country_type = colonial
			
		}
		religion = {
			has_discrimination_trait = heritage_dharmic
		}
	}

    possible = {

		OR = {
			culture_percent_country = {
				target = cu:tamil
				value > 0.25
			}
			has_law = law_type:law_multicultural
		}
		
		any_scope_state = {
			state_region = {
				is_homeland = cu:tamil
			}
		}
		has_technology_researched = egalitarianism
		cultural_acceptance_base = {
			target = cu:tamil
			value > acceptance_status_3
		}
	}

    when_taken = {

		add_primary_culture = cu:tamil
	}
	ai_chance = {
		value = 100
	}
}
