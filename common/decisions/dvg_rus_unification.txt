dvg_new_russian_capital = {
	is_shown = {
		exists = c:RUS
		this = c:RUS
		NOT = {
			has_global_variable = new_capital_rus
		}
	}

    possible = {
		has_technology_researched = railways
		has_technology_researched = modern_sewerage
		gold_reserves >= 300000
		OR = {
			owns_entire_state_region = STATE_INGRIA
			owns_entire_state_region = STATE_ROSTOV
		}
	}

    when_taken = {
		add_treasury = -300000
		set_global_variable = new_capital_rus
		trigger_event = { id = dvg_rus_iden.307}
    }

	ai_chance = {
		value = 100
	}
}
