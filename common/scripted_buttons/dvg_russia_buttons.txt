integrate_east_slavs = {
	name = integrate_east_slavs
	desc = integrate_east_slavs_desc

	visible = {
		NOT = {
			has_variable = east_slavs_in_eurasia
		}
	}
	possible = {
		AND = {
			any_scope_state = {
				state_region = {
					is_homeland = cu:russian
				}	
			}
			any_scope_state = {
				state_region = {
					is_homeland = cu:north_russian
				}	
			}
			any_scope_state = {
				state_region = {
					is_homeland = cu:cossack
				}	
			}
			any_scope_state = {
				state_region = {
					is_homeland = cu:ruthenian
				}	
			}
			any_scope_state = {
				state_region = {
					is_homeland = cu:byelorussian
				}	
			}
			radical_fraction = {
				value <= 0.25
				culture = cu:russian
			}
			radical_fraction = {
				value <= 0.25
				culture = cu:north_russian
			}
			radical_fraction = {
				value <= 0.25
				culture = cu:cossack
			}
			radical_fraction = {
				value <= 0.25
				culture = cu:ruthenian
			}
			radical_fraction = {
				value <= 0.25
				culture = cu:byelorussian
			}
		}
	}

	effect = {
		add_primary_culture = cu:eurasian
		set_variable = east_slavs_in_eurasia
		remove_primary_culture = cu:russian
		remove_primary_culture = cu:north_russian
		remove_primary_culture = cu:cossack
		remove_primary_culture = cu:ruthenian
		remove_primary_culture = cu:byelorussian
		custom_tooltip = {
			text = eurasianist_supported_start_tt
			je:je_russian_identity = {
			add_progress = { value = 50 name = russian_identity_eurasianist_progress_bar }
			}
		}
		hidden_effect = {
			every_scope_state = {
				limit = {
					state_region = {
						OR = {
							is_homeland = cu:russian
							is_homeland = cu:north_russian
							is_homeland = cu:cossack
							is_homeland = cu:ruthenian
							is_homeland = cu:byelorussian
						}
					}
				}
				state_region = {
					remove_homeland = cu:russian
					remove_homeland = cu:north_russian
					remove_homeland = cu:cossack
					remove_homeland = cu:ruthenian
					remove_homeland = cu:byelorussian
					add_homeland = cu:eurasian

				}
			}
			every_country = {
				limit = {
					OR = {
						c:RUS ?= this
						c:HRU ?= this
					}
				}
				every_scope_pop = {
					limit = {
						OR = {
							culture = cu:russian
							culture = cu:north_russian
							culture = cu:cossack
							culture = cu:ruthenian
							culture = cu:byelorussian
						}
					}	
					change_pop_culture = {
						target = cu:eurasian
						value = 1
					}
				}
			}
		}
	}

	ai_chance = {
		value = 10
	}
}
