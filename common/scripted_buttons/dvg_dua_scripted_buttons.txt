burgundy_fortress_button = {
	name = "burgundy_fortress_button"
	desc = "burgundy_fortress_button_desc"

	visible = {
		owns_entire_state_region = STATE_ILE_DE_FRANCE
		NOT = {
			has_global_variable = burgundy_fortress_global_var
		}
	}

	possible = {
		any_scope_state = {
			state_region = s:STATE_ILE_DE_FRANCE
			any_scope_building = {
				is_building_type = building_arms_industry
				level >= 5
			}
		}
	}

	effect = {
		trigger_event = { id = modernizing_naval_dua.1 popup = yes }
		add_modifier = {
			name = arsenal_of_paris_modifier
			days = long_modifier_time
		}
		set_global_variable = burgundy_fortress_global_var
		show_as_tooltip = {
			random_scope_state = {
				limit = {
					state_region = s:STATE_ILE_DE_FRANCE
				}
				add_modifier = {
					name = arsenal_of_paris_modifier
					days = very_long_modifier_time
				}
			}
		}
	}
}

rhone_iron_works_button = {
	name = "rhone_iron_works_button"
	desc = "rhone_iron_works_button_desc"

	visible = {
		owns_entire_state_region = STATE_RHONE
		NOT = {
			has_global_variable = rhone_iron_works_global_var
		}
	}

	possible = {
		any_scope_state = {
			state_region = s:STATE_RHONE
            any_scope_building = {
                is_building_type = building_iron_mine
                level >= 5
                occupancy >= 0.5
            }
            any_scope_building = {
                is_building_type = building_logging_camp
                level >= 6
                occupancy >= 0.5
            }
        }
	}

	effect = {
		trigger_event = { id = modernizing_naval_dua.2 popup = yes }
		set_global_variable = rhone_iron_works_global_var
		custom_tooltip = choose_one_of_tt
		show_as_tooltip = {
			random_scope_state = {
				limit = {
					state_region = s:STATE_RHONE
				}
				add_modifier = {
					name = rhone_iron_works_modifier
					days = very_long_modifier_time
				}
			}
			random_scope_state = {
				limit = {
					state_region = s:STATE_RHONE
				}
				if = {
					limit = {
						NOT = {
							any_scope_building = {
								is_building_type = building_steel_mill
							}
						}
					}
					create_building = {
						building = building_steel_mill
						level = 5
					}
					add_modifier = {
						name = rhone_steelworks
						days = very_long_modifier_time
					}
				}
				else = {
					add_modifier = {
						name = rhone_steelworks
						days = very_long_modifier_time
						multiplier = 1.5
					}
				}
			}
		}
	}
}
