dvg_pol_transfer_plock_button = {
	name = "dvg_pol_transfer_plock_button"
	desc = "dvg_pol_transfer_plock_button_desc"
	
	visible = {
		NOT = {
			has_variable = dvg_pol_transfer_plock_button_taken
		}
		exists = s:STATE_MAZOVIA.region_state:LIT
	}	

	possible = {
		ig:ig_landowners = {
			is_powerful = yes
			is_in_government = yes
		}
		has_insurrectionary_interest_groups = no
		custom_tooltip = {
			text = pol_maximilian_died
			c:POL = {
				ruler = {
					NOT = { has_variable = is_maximilian_jagiello }
				}
			}
		}
		has_diplomatic_pact = {
			who = c:LIT
			type = personal_union
		}
		NOT = {
			ROOT = {
				owns_entire_state_region = STATE_MAZOVIA
			}
		}
		custom_tooltip = {
			text = dvg_pol_vistula_any_works_finished_tt
			OR = {
				has_variable = dvg_pol_vistula_mouth_button_taken_var
				has_variable = dvg_pol_vistula_lublin_canal_button_var
				has_variable = dvg_pol_vistula_connecting_krakow_button_var
			}
		}
	}
	
	effect = {
		set_variable = dvg_pol_transfer_plock_button_taken
		s:STATE_MAZOVIA.region_state:LIT = {
			set_state_owner = c:POL
        }
		add_treasury = -350000
		c:LIT = {
			add_treasury = 350000
		}
	}
}

			has_variable = dvg_pol_vistula_mouth_button_taken_var
		}
	}
	
	#Yeah man you can keep it going though gdansk or you can make it go forwards or you can be biala gora's strongest warrior

		owns_entire_state_region = STATE_WEST_PRUSSIA
