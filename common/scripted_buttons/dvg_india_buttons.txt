
je_maratha_unite_aus_button = {
	name = "je_maratha_unite_aus_button"
	desc = "je_maratha_unite_aus_button_desc"

	visible = {
		always = yes
	}

	possible = {
		is_at_war = no
		has_technology_researched = nationalism
		NOT = { has_modifier = australia_test_modifier }
		AND = {
				top_overlord ?= {
				relations:root >= relations_threshold:amicable
				}
				custom_tooltip = {
				text = marathi_confederation_tt

				any_country = {
				is_at_war = no
				is_player = no
				is_adjacent_to_country = ROOT
				country_has_primary_culture = cu:marathi
				this.gdp < ROOT.gdp
				relations:ROOT >= relations_threshold:amicable
				OR = {
					is_subject = no
					is_subject_of = ROOT
					AND = {
						ROOT = { is_subject = yes }
						is_subject_of = ROOT.top_overlord
					}
				}
			}
				}
		}
			
			
		
		
	}

	cooldown = { days = short_modifier_time }

	effect = {
		custom_tooltip = unite_australia_tt

		trigger_event = { id = can_aus.3 popup = yes }


		
		add_modifier = {
			name = australia_test_modifier
			days = short_modifier_time
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				has_technology_researched = nationalism
			}
			add = 100
		}
	}
}

ceylon_presidency_button = {
	name = "ceylon_presidency_button"
	desc = "ceylon_presidency_button_desc"

	visible = { 
		any_scope_state = {
			state_region = {
				is_homeland = cu:sinhala
			}
		}
		NOT = {
			exists = c:CEY
		}
	}

	possible = {
		top_overlord = {
			relations:root >= relations_threshold:amicable
		}
		any_scope_state = {
			state_region = s:STATE_CEYLON
		}
		OR = {
			owns_entire_state_region = STATE_CEYLON
			AND = {
				exists = c:KDY
				c:KDY = {			
				is_ai = yes
				top_overlord = {
							THIS = root.top_overlord
				}
				relations:root >= relations_threshold:amicable
				this.gdp < root.gdp

				}
			}
			AND = {
				exists = c:TME
				c:TME = {			
				is_ai = yes
				top_overlord = {
							THIS = root.top_overlord
				}
				relations:root >= relations_threshold:amicable
				this.gdp < root.gdp
				}
			}
		}
	}

	effect = {
		change_tag = CEY
		top_overlord = {
			create_diplomatic_pact = {
			country = ROOT
			type = dominion
			}
		}
		if = {
			limit = {
				NOT = {
					owns_entire_state_region = STATE_CEYLON
				}
			}
			top_overlord = {
			
			every_subject_or_below = {
				limit = {
					any_scope_state = {
						state_region = s:STATE_CEYLON
						is_capital = no
					}
				}
				random_scope_state = {
					limit = {
						state_region = s:STATE_CEYLON
						is_capital = no
					}
					set_state_owner = ROOT
				}
			}
		}
		}
		
	}

	ai_chance = {
		value = 100
		 		 
	 }	
}

timor_chougie_button = {
	name = "timor_chougie_button"
	desc = "timor_chougie_button_desc"

	visible = { 
		OR = {
			any_scope_state = {
			state_region = {
				is_homeland = cu:australian
			}
			}
			any_scope_state = {
			state_region = {
				is_homeland = cu:timorese
			}
			}
		}
		
		NOT = {
			exists = c:TMC
		}
	}

	possible = {
		top_overlord = {
			relations:root >= relations_threshold:amicable
		}
		OR = {
			any_scope_state = {
			state_region = s:STATE_YOLNGU
			}
			any_scope_state = {
			state_region = s:STATE_TIMOR
			}
		}
		
		OR = {
			country_has_primary_culture = cu:timorese
			country_average_cultural_acceptance = {
			target = cu:timorese
			value >= acceptance_status_3
		}
		}
		has_technology_researched = pan-nationalism
	}

	effect = {
		change_tag = TMC
		if = {
			limit = {
				NOT = {
					country_has_primary_culture = cu:timorese
				}
			}
			add_primary_culture = cu:timorese
		}
		if = {
			limit = {
				NOT = {
					owns_entire_state_region = STATE_TIMOR
				}
			}
			top_overlord = {
			
			every_subject_or_below = {
				limit = {
					any_scope_state = {
						state_region = s:STATE_TIMOR
						is_capital = no
					}
				}
				random_scope_state = {
					limit = {
						state_region = s:STATE_TIMOR
						is_capital = no
					}
					set_state_owner = ROOT
				}
			}
			}
		}
		
	}

	ai_chance = {
		value = 100
		 		 
	 }	
}
