dvg_establish_ducal_finland_button = {
	name = "dvg_establish_ducal_finland_button_loc"
	desc = "dvg_establish_ducal_finland_button_loc_desc"
	
	#keep visible for now
	visible = {
		NOT = { #This is because you seperated the whole of Finland, not just a portion of it! They won't settle for only a meagre portion!
			#has_variable = seperated_finland_from_sweden_var
		}
		NOT = {
			has_variable = sca_established_ducal_finland_var
		}
		has_law = law_type:law_monarchy
	}
	
	#leave for now until i can understand how scoping a geo region works lmoa
	possible = {
		custom_tooltip = {
			text = sca_finland_button_owns_ducal_finland_tt
			ROOT = {
				owns_entire_state_region = STATE_UUSIMAA
				owns_entire_state_region = STATE_OSTROBOTHNIA
			}
		}
		custom_tooltip = {
			NOT = {
				any_civil_war = {
					is_civil_war_type = revolution
				}
			}
			text = sca_not_in_civil_war_tt
		}
		has_law = law_type:law_monarchy
	}
	
	effect = {
		if = {
			limit = {
				NOT = {
					has_variable = seperated_finland_from_sweden_var
				}
			}
			custom_tooltip = {
				text = sca_have_not_seperated_finland_formally_yet_tt
			}
		}
		set_variable = sca_established_ducal_finland_var
		custom_tooltip = {
			text = sca_establish_new_country_ducal_finland_tt
			create_country = {
				tag = DFN
				origin = this
				state = s:STATE_UUSIMAA.region_state:SCA
				state = s:STATE_OSTROBOTHNIA.region_state:SCA
				on_created = {
					activate_law = law_type:law_monarchy
					activate_law = law_type:law_landed_voting
					activate_law = law_type:law_subjecthood
					activate_law = law_type:law_no_colonial_affairs
					activate_law = law_type:law_slavery_banned
					c:SCA = {
						create_diplomatic_pact = {
							country = c:DFN
							type = dominion
						}
						change_relations = {
							country = c:DFN
							value = 100
						}
					}
				}
			}
		}
	}
}

dvg_expand_karelian_march_button = {
	name = "sca_expand_karelian_march_button_loc"
	desc = "sca_expand_karelian_march_button_loc_desc"

	visible = { always = yes }

	possible = {
		any_scope_state = {
			owner = ROOT
			state_region = {
				is_homeland = cu:karelian
			}
			NOT = {
				has_modifier = dvg_sca_karelian_march_modifier
			}
		}
	}
	
	effect = {
		every_scope_state = {
			limit = {
				state_region = {
					is_homeland = cu:karelian
				}
				NOT = {
					has_modifier = dvg_sca_karelian_march_modifier
				}
			}
			add_modifier = dvg_sca_karelian_march_modifier
		}
	}
	
	ai_chance = {
		value = 1
	}
}
