dvg_expand_karelian_march_button = {
	name = "sca_expand_karelian_march_button_loc"
	desc = "sca_expand_karelian_march_button_loc_desc"

	visible = { always = yes }

	possible = {
		any_scope_state = {
			owner = ROOT
			state_region = {
				is_homeland = cu:karelian
			}
			NOT = {
				has_modifier = dvg_sca_karelian_march_modifier
			}
		}
	}
	
	effect = {
		every_scope_state = {
			limit = {
				state_region = {
					is_homeland = cu:karelian
				}
				NOT = {
					has_modifier = dvg_sca_karelian_march_modifier
				}
			}
			add_modifier = dvg_sca_karelian_march_modifier
		}
	}
	
	ai_chance = {
		value = 1
	}
}
