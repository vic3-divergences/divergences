je_nov_main = {
	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	
	group = je_group_internal_affairs

	outcome = { #win
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			scope:journal_entry = {
				is_goal_complete = yes
			}
			custom_tooltip = {
				text = nov_andrei_yuriyevich_died
				ruler = {
					NOT = { has_variable = is_andrei_yuriyevich }
				}
			}
		}
		effect = {
			trigger_event = { id = dvg_nov.2 }
		}
	}

	outcome = { #lose
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				NOT = { has_law = law_type:law_presidential_republic }
				custom_tooltip = {
					text = nov_andrei_yuriyevich_died_fail
					ruler = {
						NOT = { has_variable = is_andrei_yuriyevich }
					}
					scope:journal_entry = {
						is_goal_complete = no
					}
				}
			}
		}
		effect = {
			if = {
				limit = {
					NOT = { has_law = law_type:law_presidential_republic }
				}
				trigger_event = { id = dvg_nov.4 }
			} 
			else = {
				trigger_event = { id = dvg_nov.3 }
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					ruler = { has_variable = is_andrei_yuriyevich }
					year >= 1855
				}
				ruler = { kill_character = yes }
			}
			if = {
				limit = {
					ruler = { has_variable = is_andrei_yuriyevich }
					OR = {
						AND  = {
							year >= 1846
							scope:journal_entry = {
								is_goal_complete = yes
							}
						}
					}
				}
				random_list = {
					11 = { }
					1 = {
						ruler = { kill_character = yes }
					}
				}
			}
		}
	}

	current_value = {
		value = var:nov_var
	}

	goal_add_value = {
		add = 3
	}

	progressbar = yes

	timeout = 7300

	weight = 10000
	should_be_pinned_by_default_uninvolved_or_context = yes
}
