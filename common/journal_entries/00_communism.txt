je_communism_1 = {
	icon = "gfx/interface/icons/event_icons/event_revolutionary_flag.dds"

	group = je_group_socialism

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		NOT = { has_variable = communism_1_done }
		has_technology_researched = socialism
	}

	scripted_progress_bar = communism_1_progress_bar

	possible = {
		ig:ig_trade_unions ?= {
			is_marginal = no
		}
		OR = {
			ig:ig_trade_unions ?= {
				ig_approval <= define:NPolitics|APPROVAL_THRESHOLD_UNHAPPY
				law_stance = {
					law = law_type:law_factory_councils
					value > neutral
				}
			}
			any_political_movement = {
				OR = {
					AND = {
						is_political_movement_type = movement_labor
						this.political_movement_radicalism >= define:NPolitics|RADICALISM_AGITATING_THRESHOLD
					}
					is_political_movement_type = movement_socialist
				}
			}
		}
	}

	immediate = {
		save_scope_as = root_country
		set_communism_vars = yes
		if = {
			limit = {
				has_technology_researched = socialism
				country_has_laws_in_need_of_critical_modernization = no
				NOT = {
					any_political_movement = {
						is_political_movement_type = movement_socialist
					}
				}
			}
			create_political_movement = { type = movement_socialist }
			if = {
				limit = {
					NOT = {
						any_scope_character = {
							has_role_of_type = agitator
							has_ideology = ideology:ideology_communist
						}
					}
				}
				custom_tooltip = {
					text = communist_agitator_tt
					create_character = {
						culture = ig_before_random_accepted
						is_agitator = yes
						interest_group = ig:ig_trade_unions
						ideology = ideology:ideology_communist_leader
						traits = {
							firebrand
						}
						on_created = {
							add_random_trait = personality
						}
					}
				}
			}
			random_political_movement = {
				limit = {
					is_political_movement_type = movement_socialist
				}
				save_scope_as = socialist_movement_scope
			}
		}
		else_if = {
			limit = {
				has_technology_researched = anarchism
				country_has_laws_in_need_of_critical_modernization = no
				NOT = {
					any_political_movement = {
						OR = {
							is_political_movement_type = movement_socialist
							is_political_movement_type = movement_anarchist
						}
					}
				}
			}
			create_political_movement = { type = movement_anarchist }
			if = {
				limit = {
					NOT = {
						any_scope_character = {
							has_role_of_type = agitator
							has_ideology = ideology:ideology_anarchist
						}
					}
				}
				custom_tooltip = {
					text = anarchist_agitator_tt
					create_character = {
						culture = ig_before_random_accepted
						is_agitator = yes
						interest_group = ig:ig_trade_unions
						ideology = ideology:ideology_anarchist
						traits = {
							firebrand
						}
						on_created = {
							add_random_trait = personality
						}
					}
				}
			}
			random_political_movement = {
				limit = {
					is_political_movement_type = movement_anarchist
				}
				save_scope_as = socialist_movement_scope
			}
		}
		trigger_event = { id = communism.10 popup = yes }
	}

	on_monthly_pulse = {
		random_events = {
			29 = 0
			1 = communism.7
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				custom_tooltip = {
					text = communism_progress_bar_at_full_tt
					scope:journal_entry = {
						"scripted_bar_progress(communism_1_progress_bar)" >= 100
					}
				}
				has_law_or_variant = law_type:law_council_republic
			}
		}
		effect = {
			set_variable = communism_1_done
			if = {
				limit = {
					NOT = {
						has_law_or_variant = law_type:law_council_republic
					}
				}
				trigger_event = { id = communism.1 popup = yes }
			}
			else = { # Mirrors effects of communism.1, skips je_communism_2
				if = {
					limit = {
						any_interest_group = {
							law_stance = {
								law = law_type:law_council_republic
								value > neutral
							}
						}
					}
					random_interest_group = {
						limit = {
							law_stance = {
								law = law_type:law_council_republic
								value > neutral
							}
						}
						set_variable = communist_ig_var
					}
				}
				else = {
					ig:ig_trade_unions ?= {
						set_variable = communist_ig_var
					}
				}
				if = {
					limit = {
						NOT = {
							has_journal_entry = je_state_and_revolution
						}
					}
					trigger_event = { id = communism.2 popup = yes }
				}
			}
			custom_tooltip = communism_je_1_fail_tt
			je_communism_1_cleanup = yes
		}
		event_outcome_effect_desc = {
			effect = {
				add_journal_entry = { type = je_communism_2 }
			}
		}
		event_outcome_effect_desc = {
			header = communism.1.a
			effect = {
				activate_law = law_type:law_wage_subsidies
			}
		}
		event_outcome_effect_desc = {
			header = communism.1.b
			effect = {
				activate_law = law_type:law_slavery_banned
			}
		}
		event_outcome_effect_desc = {
			header = communism.1.c
			effect = {
				activate_law = law_type:law_regulatory_bodies
			}
		}
		event_outcome_effect_desc = {
			header = communism.1.d
			effect = {
				every_interest_group = {
					limit = {
						leader = {
							OR = {
								has_ideology = ideology:ideology_vanguardist
								has_ideology = ideology:ideology_communist_leader
								has_ideology = ideology:ideology_anarchist
							}
						}
					}
					add_modifier = {
						name = communist_government_modifier
						days = normal_modifier_time
					}
				}
				add_loyalists = {
					value = medium_radicals
					strata = upper
				}
			}
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			trigger_event = { id = communism.11 popup = yes }
			custom_tooltip = communism_je_1_complete_tt
		}
		event_outcome_effect_desc = {
			header = communism.11.a
			effect = {
				add_modifier = {
					name = modifier_defeated_initial_socialist_movement
					days = normal_modifier_time
				}
				add_loyalists = {
					value = small_radicals
					strata = upper
				}
				add_loyalists = {
					value = small_radicals
					strata = middle
				}
			}
		}
		event_outcome_effect_desc = {
			header = communism.11.b
			effect = {
				add_modifier = {
					name = modifier_defeated_initial_socialist_movement_worries
					days = normal_modifier_time
				}
				add_loyalists = {
					value = medium_radicals
					strata = upper
				}
				add_loyalists = {
					value = medium_radicals
					strata = middle
				}
			}
		}
	}

	timeout = 1825

	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_anarchy = {
	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	group = je_group_socialism

	is_shown_when_inactive = {
		has_technology_researched = anarchism
	}

	possible = {
		has_law_or_variant = law_type:law_anarchy
	}

	activation_description = {
		always = yes
	}

	immediate = {
		root = {
			set_variable = { name = anarchy_starting_sol_var value = average_sol }
		}
	}

	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
					NOT = {
						any_scope_pop = {
							standard_of_living < 10
						}
					}
				}
				set_variable = anarchy_no_pop_below_sol_var
			}
			else = {
				remove_variable = anarchy_no_pop_below_sol_var
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			has_law_or_variant = law_type:law_anarchy
			has_law_or_variant = law_type:law_cooperative_ownership
			average_sol >= je_anarchy_sol_target # if you change this, also change event option in communism.2
			custom_tooltip = {
				text = je_anarchy_no_pop_below_sol_tt
				has_variable = anarchy_no_pop_below_sol_var
			}
		}
		effect = {
			trigger_event = {
				id = communism.4
				popup = yes
			}
		}
		event_outcome_effect_desc = {
			effect = {
				add_modifier = {
					name = anarchist_utopia
					days = normal_modifier_time
				}
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header

		conclusion_desc = je_standard_failure_conclusion_desc

		}
	}

	current_value = {
		value = root.average_sol
	}

	goal_add_value = {
		add = je_anarchy_sol_goal_increase
	}

	progressbar = yes

je_extraordinary_commission = {
	icon = "gfx/interface/icons/law_icons/extraordinary_commission.dds"

	group = je_group_socialism

	is_shown_when_inactive = {
		has_law_or_variant = law_type:law_council_republic
	}

	possible = {
		has_law_or_variant = law_type:law_extraordinary_commission
	}

	activation_description = {
		always = yes
	}

	immediate = {
		set_variable = { name = anti_communist_suppression_var value = 0 }
		random_political_movement = {
			limit = {
				is_political_movement_type = movement_anti_communist
			}
			save_scope_as = white_movement_scope
		}
		ig:ig_trade_unions ?= {
			save_scope_as = proletariat_ig
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					OR = {
						NOT = {
							any_political_movement = {
								is_political_movement_type = movement_anti_communist
							}
						}
						any_political_movement = {
							is_political_movement_type = movement_anti_communist
							political_movement_support < 0.05
						}
					}
				}
				change_variable = { name = anti_communist_suppression_var add = 1 }
			}
			else = {
				set_variable = { name = anti_communist_suppression_var value = 0 }
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = vanguard_custom_tooltip
				NOT = {
					any_interest_group = {
						OR = {
							is_powerful = yes
							is_in_government = yes
						}
						law_stance = {
							law = law_type:law_council_republic
							value < neutral
						}
					}
				}
			}
			legitimacy >= 75
			custom_tooltip = {
				text = no_reactionary_conservative_liberal_agitators_tt
				NOT = {
					any_scope_character = {
						OR = {
							has_reactionary_ideology = yes
							has_conservative_ideology = yes
							has_liberal_ideology = yes
						}
						has_role_of_type = agitator
					}
				}
			}
			custom_tooltip = {
				text = has_suppressed_white_movement_for_some_years_tt
				var:anti_communist_suppression_var >= 24
				OR = {
					NOT = {
						any_political_movement = {
							is_political_movement_type = movement_anti_communist
						}
					}
					any_political_movement = {
						is_political_movement_type = movement_anti_communist
						political_movement_support < 0.05
					}
				}
			}
			custom_tooltip = {
				text = is_not_in_civil_war_tt
				any_civil_war = {
					count = 0
				}
			}
		}
		effect = {
			trigger_event = { id = communism.5 popup = yes }
		}
		event_outcome_effect_desc = {
			effect = {
				add_modifier = {
					name = great_purge_modifier
					days = normal_modifier_time
				}
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = { has_law_or_variant = law_type:law_council_republic }
		}
		effect = {
			je_extraordinary_commission_cleanup = yes
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {
				has_law_or_variant = law_type:law_extraordinary_commission
			}
		}
		effect = {
			je_extraordinary_commission_cleanup = yes
		}
	}

	on_weekly_pulse = {
		events = {
			communism.3
		}
