
je_ameriga_unite = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	group = je_group_internal_affairs

	is_shown_when_inactive = {
		exists = c:AME
		this = c:AME
	}

    possible = {
		has_technology_researched = pan-nationalism
		OR = {
			country_has_primary_culture = cu:antarctican
			country_has_primary_culture = cu:equinoxien
			country_has_primary_culture = cu:lotharian
			country_has_primary_culture = cu:atlantian
			AND = {
				has_variable = blt_pursue_ameriga
				country_has_primary_culture = cu:beornian
			}
		}
		OR = {
			is_subject = no
			aggressive_diplomatic_plays_permitted = yes
		}
	}

	outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = je_standard_completion_conclusion_desc
        trigger = {
			is_diplomatic_play_committed_participant = no
			any_scope_state = {
			count >= 5
			AND = {	
				region = sr:region_brazil
				owner = ROOT
			}
			}
			any_scope_state = {
				count >= 5
				AND = {	
					region = sr:region_la_plata
					owner = ROOT
				}
			}
			any_scope_state = {
				count >= 5
				AND = {	
					region = sr:region_gran_colombia
					owner = ROOT
				}
			}
			any_scope_state = {
				count >= 5
				AND = {	
					region = sr:region_andes
					owner = ROOT
				}
			}
		}
        effect = {
			trigger_event = { id = dvg_ameriga.3 popup = yes }
		}
        event_outcome_desc = {} # complex desc like a status_desc field
    }

    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {
				AND = {
					any_interest_group = { is_insurrectionary = yes }
					any_scope_state = {
						count >= 5
					}
				}
				AND = {	
					region = sr:region_brazil
					turmoil > 0.9
					owner = ROOT
					}
				any_scope_state = {
				count >= 3
				AND = {	
					region = sr:region_la_plata
					owner = ROOT
					turmoil > 0.9
				}
				}
				average_sol <= 7
				has_modifier = declared_bankruptcy
			}
        effect = {
			trigger_event = { id = dvg_ameriga.4 popup = yes }
		}
    }

    outcome = {
        flags = { hidden invalid_outcome }
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {
			has_global_variable = amerigan_culture
		}
        effect = {}
    }

	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_amerigan_nation_building = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_historical_content

	is_shown_when_inactive = {
		NOT = {
			country_has_primary_culture = cu:amazonien
		}
		exists = c:AME
		c:AME = ROOT
	}
	possible = {
		has_technology_researched = pan-nationalism
		OR = {
			NOT = {
				exists = c:AMA
			}
			NOT = {
				exists = c:LOT
			}
			NOT = {
				exists = c:ATL
			}
			NOT = {
				exists = c:EQX
			}
		}
	}

	immediate = {
		
		cu:lotharian = {
			save_scope_as = atlantian_culture
		}
		cu:atlantian = {
			save_scope_as = lotharian_culture
		}
		cu:equinoxien = {
			save_scope_as = equinoxien_culture
		}
		cu:antarctican = {
			save_scope_as = antarctican_culture
		}
		cu:amerigan = {
			save_scope_as = amerigan_culture
		}
	}

	outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = je_standard_completion_conclusion_desc
        trigger = {
			has_technology_researched = pan-nationalism
			custom_tooltip = {
				text = je_amerigan_nation_building_integrate_equinoxien_complete_tt
				has_variable = equinoxiens_integrated
			}
			custom_tooltip = {
				text = je_amerigan_nation_building_integrate_lotharian_complete_tt
				has_variable = lotharians_integrated
			}
			custom_tooltip = {
				text = je_amerigan_nation_building_integrate_atlantian_complete_tt
				has_variable = atlantians_integrated
			}
			custom_tooltip = {
				text = je_amerigan_nation_building_integrate_antarctican_complete_tt
				OR = {
					has_variable = antarcticans_integrated
					NOT = {
						any_state_region = {
							is_homeland = cu:antarctican
						}
					}
				}
			}			
		}
        effect = {
			trigger_event = { id = dvg_ameriga.7 }
			set_global_variable = amerigan_culture
			add_loyalists = {
				value = 0.20
			}
			add_modifier = {
				name = dvg_nation_built
			}
		}
        event_outcome_desc = {} # complex desc like a status_desc field
    }

    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {
			AND = {
				exists = c:AMA
				exists = c:LOT
				exists = c:ATL
				exists = c:EQX
			}
		}
        effect = {}
    }

	scripted_button = je_amerigan_nation_building_integrate_equinoxien
	scripted_button = je_amerigan_nation_building_integrate_lotharian
	scripted_button = je_amerigan_nation_building_integrate_atlantian
	scripted_button = je_amerigan_nation_building_integrate_antarctican

	weight = 100

	should_be_pinned_by_default_uninvolved_or_context = yes
}
