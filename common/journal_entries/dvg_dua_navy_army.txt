je_dua_army = {
	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	should_be_pinned_by_default_uninvolved_or_context = yes

	group = je_group_internal_affairs

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "red_positive"
		trigger = {
			army_size >= 150
			has_technology_researched = shell_gun
			has_technology_researched = percussion_cap
			has_technology_researched = rifling
			NOR = {
				any_scope_building = {
					is_building_type = building_barrack				
					OR = {
						building_has_goods_shortage = yes
						has_active_production_method = pm_general_training
					}
				}
				any_military_formation = {
					filter = {
						is_army = yes
					}
					any_combat_unit = {
						has_unit_type = unit_type:combat_unit_type_mobile_artillery
					}
					percent >= 0.25
				}
			}
			calc_true_if = {
				amount > 4
				NOR = {
					has_law = law_type:law_states_general
				}
				NOR = {
					has_law = law_type:law_peasant_levies
				}
				NOR = {
					has_law = law_type:law_subjecthood
				}
				NOR = {
					has_law = law_type:law_serfdom
				}
			}
			bureaucracy >= 0
		}
		effect = {
			remove_modifier = dvg_outmoded_dua_army
		}
	}

	transferable = yes

	should_be_pinned_by_default_uninvolved_or_context = yes

	weight = 1000
}

je_nayezpaspeur = {
	icon = "gfx/interface/icons/event_icons/bicorne_hat.dds"

	group = je_group_internal_affairs

	is_shown_when_inactive = {
		c:DUA ?= THIS
	}

	possible = {
		country_rank >= rank_value:minor_power
		any_scope_character = {
			has_role_of_type = admiral
		}
		OR = {
			has_technology_researched = dreadnought_tech
			has_technology_researched = sea_lane_strategies
		}
	}

	immediate = {
		set_variable = {
			name = dua_navy_var
			value = 0
		}
		# startup event
		trigger_event = {
			id = dua_navy.1
			popup = yes
		}
	}

	on_monthly_pulse = {
		random_events = {
			100 = 0
			5 = dua_navy.2 # Navy complains that it's not as powerful as it should be
			5 = dua_navy.3 # navy is strong
		}
		effect = {
			if = {
				limit = {
					NOT = {
						any_country = {
							naval_power_projection > root.naval_power_projection
							count > 10
						}
					}
					NOT = {
						any_country_in_south_america = {
							naval_power_projection > root.naval_power_projection
						}
					}
				}
				change_variable = {
					name = dua_navy_var
					add = 1
				}
			}
		}
	}

	is_progressing = {
		NOT = {
			any_country = {
				naval_power_projection > root.naval_power_projection
				count > 10
			}
		}
		NOT = {
			any_country_in_south_america = {
				naval_power_projection > root.naval_power_projection
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			any_scope_character = {
				has_role_of_type = admiral
			}
			has_law_or_variant = law_type:law_professional_navy
			scope:journal_entry = {
				is_goal_complete = yes
			}
			NOT = {
				any_country = {
				naval_power_projection > root.naval_power_projection
				count > 10
			}
			}
			custom_tooltip = {
				text = best_european_navy
				NOT = {
					any_country_in_west_europe = {
						naval_power_projection > root.naval_power_projection
				}
			}
		}
		effect = {
			trigger_event = {
			id = dua_navy.4
			popup = yes
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			country_rank < rank_value:minor_power
		}
		effect = {
			je_dua_navy_cleanup = yes
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			je_brazil_navy_cleanup = yes
		}
	}

	timeout = 25550

	current_value = {
		value = root.var:dua_navy_var
	}

	goal_add_value = {
		add = 480
	}

	progressbar = yes

	weight = 100

	can_revolution_inherit = no
	transferable = no
		
	should_be_pinned_by_default_uninvolved_or_context = yes
	}
}
