je_romanian_unification = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	should_be_pinned_by_default_uninvolved_or_context = yes

	group = je_group_foreign_affairs

	is_shown_when_inactive = {
		country_has_primary_culture = cu:romanian
	}

	possible = {
		NOR = {
			has_journal_entry = je_moldavian_troubles 
			has_journal_entry = je_national_liberation_government	
		}
		OR = {
			c:WAL = this
			c:MOL = this
			c:ROM = this
		}
	}

	outcome = { #win
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			calc_true_if = {
				amount >= 9
				owns_entire_state_region = STATE_BANAT
				owns_entire_state_region = STATE_WALLACHIA
				owns_entire_state_region = STATE_DOBRUDJA
				owns_entire_state_region = STATE_MOLDAVIA
				owns_entire_state_region = STATE_BESSARABIA
				owns_entire_state_region = STATE_NORTHERN_TRANSYLVANIA
				owns_entire_state_region = STATE_SOUTHERN_TRANSYLVANIA
				owns_entire_state_region = STATE_BUKOVINA
				owns_entire_state_region = STATE_BEKES
				owns_entire_state_region = STATE_VOJVODINA
				owns_entire_state_region = STATE_EAST_GALICIA
				owns_entire_state_region = STATE_KIEV
				owns_entire_state_region = STATE_CHERSON
			}
		}
		effect = {
			if = {
				limit = {
					owns_entire_state_region = STATE_VOJVODINA
				}
				s:STATE_VOJVODINA = {
					add_homeland = cu:romanian
				}
			}
			if = {
				limit = {
					owns_entire_state_region = STATE_EAST_GALICIA
				}
				s:STATE_EAST_GALICIA = {
					add_homeland = cu:romanian
				}
			}
			if = {
				limit = {
					owns_entire_state_region = STATE_KIEV
				}
				s:STATE_KIEV = {
					add_homeland = cu:romanian
				}
			}
			if = {
				limit = {
					owns_entire_state_region = STATE_CHERSON
				}
				s:STATE_CHERSON = {
					add_homeland = cu:romanian
				}
			}
			trigger_event = { id = dvg_romania.4 }
			add_loyalists = {
				value = 0.25
				culture = cu:romanian
			}
			set_global_variable = dacia_united
			}
	}
}
je_moldavian_troubles = {
	icon = "gfx/interface/icons/event_icons/event_broken_crown.dds"
	should_be_pinned_by_default_uninvolved_or_context = yes

	group = je_group_foreign_affairs

	outcome = { #win
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			has_law = law_type:law_national_liberation
			OR = {
				ig:ig_armed_forces = {
					is_powerful = yes
					is_in_government = yes
				}
				ig:ig_petty_bourgeoisie = {
					is_powerful = yes
					is_in_government = yes
				}
			}
		}
		effect = {
			ruler = {
				kill_character = yes
			}
			c:WAL  ={
				trigger_event = {id=dvg_romania.32 days = 7}
			}
		}
	}
	outcome = { #lose
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {has_law = law_type:law_monarchy}
		}
		effect = {
			add_radicals = {
				value = 0.25
			}
		}
	}
	timeout = 36500

	outcome = {
        header = JOURNAL_ENTRY_TIMEOUT
        effect_header = JOURNAL_ENTRY_ON_TIMEOUT
        flags = { timeout }
        conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
        effect = {
			ruler = {
				kill_character = yes
			}
			trigger_event = {id=dvg_romania.31 days = 7}
			ig:ig_armed_forces = {
				set_interest_group_name = ig_armed_forces
				add_ideology = ideology_loyalist
				remove_ideology = ideology_national_liberal
			}
			ig:ig_petty_bourgeoisie= {
				set_interest_group_name = ig_petty_bourgeoisie
				add_ideology = ideology_reactionary
				remove_ideology = ideology_national_liberal
				add_character_ideology = ideology_traditionalist
			}
		}
	}
}
