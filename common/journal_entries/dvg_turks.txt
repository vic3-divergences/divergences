
je_janissary = {

 icon = "gfx/interface/icons/event_icons/event_military.dds"
 group = je_group_foreign_affairs

 

 is_shown_when_inactive = {
	exists = c:JNR
	c:JNR = ROOT
 }

 possible = {
	is_diplomatic_play_enemy_of = c:TUR
	is_at_war = yes
 }

 immediate = {
	c:TUR = {
		save_scope_as = scope_TUR
		ruler = {
			save_scope_as = tur_ruler_scope
		}
	}
 }
 
outcome = { #win
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			owns_entire_state_region = STATE_EASTERN_THRACE
			owns_entire_state_region = STATE_ANKARA
			is_diplomatic_play_committed_participant = no
		}
		effect = {
			trigger_event = {
				id = dvg_ottomans_reclaim.5
			}
			set_global_variable = janissary_victory
		}
}

   weight = 100 
}
