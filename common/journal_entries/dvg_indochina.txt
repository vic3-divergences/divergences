		has_technology_researched = nationalism
	}

	immediate = {

	on_monthly_pulse = {
		fallback = on_yearly_events

	}

	complete = {
		calc_true_if = {
			amount > 4

			has_technology_researched = pan-nationalism

			NOT = {
				has_law = law_type:law_subjecthood
			}

			is_country_type = recognized

			AND = {
				owns_entire_state_region = STATE_ARAKAN

				}

						is_homeland = cu:shan
					}
				}

				}
			}

				radical_fraction = {
				culture = cu:burmese
				value > 0.33
				}

					state_region = {
						is_homeland = cu:burmese
					}
					radical_fraction = {
						value > 0.33
					}
				}
				cu:burmese = {
				culture_current_fervor > 50
				}
			}
			AND = {
				radical_fraction = {
				culture = cu:karen
				value > 0.33
				}
				any_scope_state = {
					state_region = {
						is_homeland = cu:karen
					}
					radical_fraction = {
						value > 0.33
					}
				}

				culture_current_fervor > 50

				culture = cu:kachin
