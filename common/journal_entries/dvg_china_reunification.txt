je_china_reunification = {
	icon = "gfx/interface/icons/invention_icons/nationalism.dds"

	modifiers_while_active = { dvg_chinese_bureau }

	group = je_group_foreign_affairs

	scripted_button = annex_chinese_subjects
	scripted_button = china_unify_play

    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = je_standard_completion_conclusion_desc
        trigger = {
			calc_true_if = {
				amount >= 18
				country_or_subject_owns_entire_state_region = STATE_GANSU
				country_or_subject_owns_entire_state_region = STATE_SHENGJING
				country_or_subject_owns_entire_state_region = STATE_BEIJING
				country_or_subject_owns_entire_state_region = STATE_SHANXI
				country_or_subject_owns_entire_state_region = STATE_XIAN

				country_or_subject_owns_entire_state_region = STATE_ZHILI
				country_or_subject_owns_entire_state_region = STATE_SHANDONG
				country_or_subject_owns_entire_state_region = STATE_JIANGSU
				country_or_subject_owns_entire_state_region = STATE_NORTHERN_ANHUI
				country_or_subject_owns_entire_state_region = STATE_NANJING
				country_or_subject_owns_entire_state_region = STATE_SUZHOU
				country_or_subject_owns_entire_state_region = STATE_SOUTHERN_ANHUI
				country_or_subject_owns_entire_state_region = STATE_ZHEJIANG
				country_or_subject_owns_entire_state_region = STATE_FORMOSA
				country_or_subject_owns_entire_state_region = STATE_FUJIAN
				country_or_subject_owns_entire_state_region = STATE_SHAOZHOU
				country_or_subject_owns_entire_state_region = STATE_JIANGXI
				country_or_subject_owns_entire_state_region = STATE_EASTERN_HUBEI
				country_or_subject_owns_entire_state_region = STATE_WESTERN_HUBEI
				country_or_subject_owns_entire_state_region = STATE_HUNAN
				country_or_subject_owns_entire_state_region = STATE_GUANGXI
				country_or_subject_owns_entire_state_region = STATE_GUANGDONG
				country_or_subject_owns_entire_state_region = STATE_GUIZHOU
				country_or_subject_owns_entire_state_region = STATE_CHONGQING
				country_or_subject_owns_entire_state_region = STATE_SICHUAN
				country_or_subject_owns_entire_state_region = STATE_YUNNAN
			}			
		}
        effect = {
			trigger_event = { id = dvg_china.1 } # China Reunification

			add_modifier = {
				name = dvg_chinese_bureau
				years = 15
				is_decaying = yes
			}			
		}
        event_outcome_desc = {} # complex desc like a status_desc field
    }

    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {
			has_global_variable = china_reunified
		}
        effect = {
			trigger_event = { id = dvg_china.2 } # China Reunification Failed
		}
    }

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					any_scope_state = {
						state_region = s:STATE_GANSU
						is_isolated_from_market = yes
					}
					NOT = {
						exists = c:XIB
					}
				}
				random_scope_state = {
					limit = {
						state_region = s:STATE_GANSU
					}
					set_state_owner = c:XIB
				}
			}
			if = {
				limit = {
					any_scope_state = {
						state_region = s:STATE_SICHUAN
						is_isolated_from_market = yes
					}
					NOT = {
						exists = c:SIH
					}
				}
				random_scope_state = {
					limit = {
						state_region = s:STATE_SICHUAN
					}
					set_state_owner = c:SIH
				}
			}
			if = {
				limit = {
					any_scope_state = {
						state_region = s:STATE_NANJING
						is_isolated_from_market = yes
					}
					NOT = {
						exists = c:NAN
					}
				}
				random_scope_state = {
					limit = {
						state_region = s:STATE_NANJING
					}
					set_state_owner = c:NAN
				}
			}
			if = {
				limit = {
					any_scope_state = {
						state_region = s:STATE_YUNNAN
						is_isolated_from_market = yes
					}
					NOT = {
						exists = c:YUN
					}
				}
				random_scope_state = {
					limit = {
						state_region = s:STATE_YUNNAN
					}
					set_state_owner = c:YUN
				}
			}
			if = {
				limit = {
					any_scope_state = {
						state_region = s:STATE_CHONGQING
						is_isolated_from_market = yes
					}
					
				}
				random_scope_state = {
					set_state_owner = c:YUN
				}
			}
			if = {
				limit = {
					NOT = {
						has_technology_researched = nationalism
					}
					has_technology_researched = mass_communication
					year >= 1853
				}
				add_technology_researched = nationalism
			}
		}
	}

	weight = 100

	should_be_pinned_by_default_uninvolved_or_context = yes
}
