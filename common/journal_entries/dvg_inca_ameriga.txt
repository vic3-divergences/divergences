je_inca_lotharingia = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_historical_content

	is_shown_when_inactive = {
		exists = c:LOT
		exists = c:TWT
		c:TWT = ROOT
		has_global_variable = anti_inca_coalition
	}
	possible = {
		exists = c:LOT
		is_diplomatic_play_enemy_of = c:LOT
	}

	outcome = { #win
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = lot_existence_tt
				NOT = { exists = c:LOT }
			}
		}
		effect = {
			if = {
				limit = {
					is_country_type = unrecognized
				}
				set_country_type = recognized
			}
			
			trigger_event = { id = dvg_inca_ameriga.5} 

			set_global_variable = lotharingia_shatters
		}
	}

	outcome = { #lose
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = twt_capitulation_tt
				has_truce_with = c:LOT
			}
		}
		effect = {
			trigger_event = { id = dvg_inca_ameriga.9 }

			every_diplomatic_play = {
				limit = {
					OR = {
						initiator_is = c:LOT
						initiator_is = c:AMA
					}
				}
				end_play = yes
			}
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			trigger_event = { id = dvg_inca_ameriga.8 }
		}
	}

	timeout = 1200

	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
}
