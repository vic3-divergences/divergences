je_hungarian_crisis = {
	icon = "gfx/interface/icons/event_icons/event_hungary_in_jail.dds"
	should_be_pinned_by_default_uninvolved_or_context = yes

	group = je_group_crises

	immediate = {
		set_variable = {
			name = dvg_hungarian_crisis_boh_power
			value = 0
		}

		set_variable = {
			name = dvg_hungarian_crisis_trs_power
			value = 0
		}

		set_variable = {
			name = dvg_hungarian_crisis_pom_power
			value = 0
		}

	}
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = je_hungarian_crisis_complete_tooltip
				has_global_variable = dvg_hungarian_crisis_complete
			}
		}
		effect = {
			custom_tooltip = je_hungarian_crisis_on_complete_tooltip
			hidden_effect = {
				if = {
					limit = {
						this = c:BOH
						var:dvg_hungarian_crisis_pom_power >= 6 }
		
					every_country = {
						limit = {
							has_variable = dvg_attend_hungarian_conference
						}
						trigger_event = { id = dvg_hungary.61 days = 10 }
						
					}
		
				} else_if = {
					
					limit = {
						this = c:BOH
					}
					every_country = {
						limit = {
							has_variable = dvg_attend_hungarian_conference
						}
						trigger_event = { id = dvg_hungary.62 days = 10 }
						
					}

					c:BOH = {
						trigger_event = { id = dvg_bohemia.10 days = 72 }
						
					}
		
				}
		
				if = {#jagiellon hungary
					limit = { 
						this = c:BOH
						AND = {
							var:dvg_hungarian_crisis_boh_power >= 6 
							var:dvg_hungarian_crisis_boh_power > var:dvg_hungarian_crisis_trs_power
						}
					}
		
					every_country = {
						limit = {
							has_variable = dvg_attend_hungarian_conference
						}
						trigger_event = { id = dvg_hungary.71 days = 10 }
						
					}
		
				} 
				else_if = {#kemeny hungary

					limit = {
						this = c:BOH
						AND = {
							var:dvg_hungarian_crisis_trs_power >= 6
							var:dvg_hungarian_crisis_trs_power > var:dvg_hungarian_crisis_boh_power
						}
					}
		
					every_country = {
						limit = {
							has_variable = dvg_attend_hungarian_conference
						}
						trigger_event = { id = dvg_hungary.72 days = 10 }
						
					}
				}
				else_if = {#independent

					limit = {
						this = c:BOH
					}
		
					every_country = {
						limit = {
							has_variable = dvg_attend_hungarian_conference
						}
						trigger_event = { id = dvg_hungary.73 days = 10 }
						
					}
				}
			}
		}
	}
	timeout = 3650
	
	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			custom_tooltip = {
				text = je_hungarian_crisis_timeout_tooltip
				if = {
					
					limit = {
						this = c:BOH
					}
					every_country = {
						limit = {
							has_variable = dvg_attend_hungarian_conference
						}
						trigger_event = { id = dvg_hungary.62 days = 10 }
						
					}

					c:BOH = {
						trigger_event = { id = dvg_bohemia.10 days = 72 }
						
					}
		
				}
				if = {#independent

					limit = {
						this = c:BOH
					}
		
					every_country = {
						limit = {
							has_variable = dvg_attend_hungarian_conference
						}
						trigger_event = { id = dvg_hungary.73 days = 10 }
						
					}
				}
			}
		}
	}
}
je_hungarian_interregnum = {
	icon = "gfx/interface/icons/event_icons/event_empty_chair.dds"
	should_be_pinned_by_default_uninvolved_or_context = yes
	group = je_group_internal_affairs

	modifiers_while_active = {
		diplomatic_plays_blocked
	}
	immediate = {
	}
	
	outcome = { #win
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				custom_tooltip = {
					text = interregnum_over_tt
					has_global_variable = hungarian_constitutional_crisis_over
				}
				AND = {
					owns_entire_state_region = STATE_CENTRAL_HUNGARY
					owns_entire_state_region = STATE_TRANSDANUBIA
					owns_entire_state_region = STATE_SLOVAKIA
				}
			}
		}
		effect = {
			add_loyalists = {
				value = 0.05
			}
			add_journal_entry = {
				type = je_hungarian_expansion
			}
			set_capital = STATE_CENTRAL_HUNGARY
		}
	}
	outcome = { #lose
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {has_law = law_type:law_monarchy}
		}
		effect = {
			add_radicals = {
				value = 0.20
			}
			add_modifier = {
				name = dvg_overburdened_bureaucracy 
				months = 48
				is_decaying = yes
			}
	
		}
	}
}

je_balkan_armies = {
	icon = "gfx\interface\icons\alert_icons\suboptimal_army.dds"
	should_be_pinned_by_default_uninvolved_or_context = yes
	group = je_group_internal_affairs

	immediate = {
	}
	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
					NOT = { has_law_or_variant = law_type:law_peasant_levies }
				
				}
				set_variable = militias_assimilated
			}
		}
	}
	status_desc = {
			triggered_desc = {
				desc = je_balkan_armies_status
				trigger = {
					always = yes
				}
			}
			triggered_desc = {
				desc = status_mercenaries_paid_succesfully
				trigger = {
					has_variable = mercenaries_paid
				}
			}
			triggered_desc = {
				desc = status_voluteers_integrated
				trigger = {
					has_variable = voluteers_integrated
				}
			}
			triggered_desc = {
				desc = status_militias_assimilated
				trigger = {
					has_variable = militias_assimilated
				}
			}
	}

	outcome = { #win
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = mercenaries_paid_tt
				has_variable = mercenaries_paid
			}
			custom_tooltip = {
				text = voluteers_integrated_tt
				has_variable = voluteers_integrated
			}
			custom_tooltip = {
				text = militias_assimilated_tt
				has_variable = militias_assimilated
			}
		}
		effect = {
			add_loyalists = {
				value = 0.15
				pop_type = soldiers
			}
			add_loyalists = {
				value = 0.1
				pop_type = officers
			}
		}
	}
	scripted_button = hungary_pay_mercs
	scripted_button = hungary_assimilate_volunteers
	
}
