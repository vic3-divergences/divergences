je_hungarian_crisis = {
	icon = "gfx/interface/icons/event_icons/event_hungary_in_jail.dds"
	should_be_pinned_by_default_uninvolved_or_context = yes

	group = je_group_crises

	immediate = {
		set_variable = {
			name = dvg_hungarian_crisis_boh_power
			value = 0
		}

		set_variable = {
			name = dvg_hungarian_crisis_trs_power
			value = 0
		}

		set_variable = {
			name = dvg_hungarian_crisis_pom_power
			value = 0
		}

	}

	complete = {
		custom_tooltip = {
			text = je_hungarian_crisis_complete_tooltip
			has_global_variable = dvg_hungarian_crisis_complete
		}
	}
	timeout = 3650
	on_timeout = {
		custom_tooltip = {
			text = je_hungarian_crisis_timeout_tooltip
			if = {
				
				limit = {
					this = c:BOH
				}
				every_country = {
					limit = {
						has_variable = dvg_attend_hungarian_conference
					}
					trigger_event = { id = dvg_hungary.62 days = 10 }
					
				}

				c:BOH = {
					trigger_event = { id = dvg_bohemia.10 days = 72 }
					
				}
	
			}
			if = {#independent

				limit = {
					this = c:BOH
				}
	
				every_country = {
					limit = {
						has_variable = dvg_attend_hungarian_conference
					}
					trigger_event = { id = dvg_hungary.73 days = 10 }
					
				}
			}
		}
	}

	on_complete = {
		custom_tooltip = je_hungarian_crisis_on_complete_tooltip
		hidden_effect = {
			if = {
				limit = {
					this = c:BOH
					var:dvg_hungarian_crisis_pom_power >= 6 }
	
				every_country = {
					limit = {
						has_variable = dvg_attend_hungarian_conference
					}
					trigger_event = { id = dvg_hungary.61 days = 10 }
					
				}
	
			} else_if = {
				
				limit = {
					this = c:BOH
				}
				every_country = {
					limit = {
						has_variable = dvg_attend_hungarian_conference
					}
					trigger_event = { id = dvg_hungary.62 days = 10 }
					
				}

				c:BOH = {
					trigger_event = { id = dvg_bohemia.10 days = 72 }
					
				}
	
			}
	
			if = {#jagiellon hungary
				limit = { 
					this = c:BOH
					AND = {
						var:dvg_hungarian_crisis_boh_power >= 6 
						var:dvg_hungarian_crisis_boh_power > var:dvg_hungarian_crisis_trs_power
					}
				}
	
				every_country = {
					limit = {
						has_variable = dvg_attend_hungarian_conference
					}
					trigger_event = { id = dvg_hungary.71 days = 10 }
					
				}
	
			} 
			else_if = {#kemeny hungary

				limit = {
					this = c:BOH
					AND = {
						var:dvg_hungarian_crisis_trs_power >= 6
						var:dvg_hungarian_crisis_trs_power > var:dvg_hungarian_crisis_boh_power
					}
				}
	
				every_country = {
					limit = {
						has_variable = dvg_attend_hungarian_conference
					}
					trigger_event = { id = dvg_hungary.72 days = 10 }
					
				}
			}
			else_if = {#independent

				limit = {
					this = c:BOH
				}
	
				every_country = {
					limit = {
						has_variable = dvg_attend_hungarian_conference
					}
					trigger_event = { id = dvg_hungary.73 days = 10 }
					
				}
			}
		}
	}
}
je_hungarian_interregnum = {
	icon = "gfx/interface/icons/event_icons/event_empty_chair.dds"
	should_be_pinned_by_default_uninvolved_or_context = yes
	group = je_group_internal_affairs

	is_shown_when_inactive = {
	}
	modifiers_while_active = {
		diplomatic_plays_blocked
	}
	possible = {
	}

	immediate = {
	}

	complete = {
		custom_tooltip = {
			text = interregnum_over_tt
			has_global_variable = hungarian_constitutional_crisis_over
		}
		
	}

	on_complete = {
		add_loyalists = {
			value = 0.05
		}
		add_journal_entry = {
			type = je_hungarian_expansion
		}
		set_capital = STATE_CENTRAL_HUNGARY
	}

	fail = {
		NOT = {has_law = law_type:law_monarchy}
		
	}
	on_fail = {
		add_radicals = {
			value = 0.20
		}
		add_modifier = {
			name = dvg_overburdened_bureaucracy 
			months = 48
			is_decaying = yes
		}
		
	}
	
}
je_hungarian_expansion = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	should_be_pinned_by_default_uninvolved_or_context = yes
	group = je_group_internal_affairs

	immediate = {
		s:STATE_CROATIA = {
			add_claim = c:HUN
		}
		s:STATE_SLAVONIA = {
			add_claim = c:HUN
		}
		s:STATE_BOSNIA = {
			add_claim = c:HUN
		}
		s:STATE_BANAT = {
			add_claim = c:HUN
		}
		s:STATE_VOJVODINA = {
			add_claim = c:HUN
		}
	}

	complete = {
		OR = {
			custom_tooltip = {
				text = croatian_lands_tt
				country_or_subject_owns_entire_state_region = STATE_CROATIA
				country_or_subject_owns_entire_state_region = STATE_SLAVONIA
				country_or_subject_owns_entire_state_region = STATE_BOSNIA
			}
			custom_tooltip = {
				text = croatian_independence_tt
				has_global_variable = croatia_je_done
			}
		}
		custom_tooltip = {
			text = banat_lands_tt
			country_or_subject_owns_entire_state_region = STATE_BANAT
			country_or_subject_owns_entire_state_region = STATE_VOJVODINA
		}
		
	}

	on_complete = {
		#trigger_event = {id = dvg_hungary.78 days = 7}
		add_loyalists = {
			value = 0.15
		}
	}

	fail = {
		c:CRO = {
			country_rank > rank_value:minor_power
		}
		c:BSG = {
			country_rank > rank_value:minor_power
		}
	}
	on_fail = {
		add_radicals = {
			value = 0.5
		}
		
	}
	scripted_button = hun_croatian_button
	scripted_button = hun_banat_button
}
