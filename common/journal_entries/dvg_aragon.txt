je_conquest_of_algeria = {
	icon = "gfx/interface/icons/event_icons/bicorne_hat.dds"

	group = je_group_historical_content

	activation_description = {
		always = yes
	}

	is_shown_in_lobby = {
		c:ARN = THIS
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			country_or_subject_owns_entire_state_region = STATE_ORAN
			country_or_subject_owns_entire_state_region = STATE_ALGIERS
			country_or_subject_owns_entire_state_region = STATE_CONSTANTINE
			trigger_if = {
				limit = {
					exists = c:TUG
				}
				c:TUG ?= {
					OR = {
						is_country_alive = no
						is_subject_of = root
					}
				}
			}
		}
		effect = {
			custom_tooltip = je_conquest_of_algeria_tt
			trigger_event = algeria_events.1
			show_as_tooltip = {
				every_scope_state = {
					limit = {
						OR = {
							state_region = s:STATE_ORAN
							state_region = s:STATE_ALGIERS
							state_region = s:STATE_CONSTANTINE
						}
					}
					add_modifier = {
						name = crushed_algerian_resistance
						days = very_long_modifier_time
					}
				}
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = { has_or_is_contesting_state_in_state_region = STATE_ORAN }
			NOT = { has_or_is_contesting_state_in_state_region = STATE_ALGIERS }
			NOT = { has_or_is_contesting_state_in_state_region = STATE_CONSTANTINE }
		}
		effect = {
			trigger_event = algeria_events.3
		}
	}

	on_monthly_pulse = {
		random_events = {
			90 = 0
			10 = algeria_events.5 # Raid Event
		}
	}

	transferable = no


	should_be_pinned_by_default_uninvolved_or_context = yes

	weight = 5000
}
