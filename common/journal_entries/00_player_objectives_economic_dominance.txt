je_objective_expand_goods_production = {
		
	should_be_pinned_by_default_uninvolved_or_context = yes
	icon = "gfx/interface/icons/objectives/economic_dominance.dds"

	group = je_group_objectives

	activation_description = {
		custom_tooltip = je_objective_expand_goods_production_activation_tt
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			scope:journal_entry = { is_goal_complete = yes }
			scope:target = {
				cash_reserves_ratio > 0.25
				weekly_profit > 0
				occupancy > 0.75
				is_subsidized = no
			}
		}
		effect = {
			hidden_effect = {
				owner = {
					start_tutorial_lesson = {
						tutorial_lesson = expand_goods_production_complete
						journal_entry = je:je_objective_expand_goods_production
					}
				}
			}
			je_objective_expand_goods_production_cleanup = yes
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = je_objective_expand_goods_production_invalid_tooltip
				OR = {
					NOT = { exists = scope:target }
					NOT = { scope:target.owner = root }
				}
			}
		}
		effect = {
			remove_active_objective_subgoal = sg_expand_goods_production
			je_objective_expand_goods_production_cleanup = yes
		}
	}

	current_value = {
		value = scope:target.level
	}

	goal_add_value = {
		value = 1
	}

	is_progressing = {
		exists = scope:target
		scope:target = { is_under_construction = yes }
	}

	weight = 1000
	transferable = yes
}

je_specialized_inputs = {
		
	should_be_pinned_by_default_uninvolved_or_context = yes
	icon = "gfx/interface/icons/objectives/economic_dominance.dds"

	group = je_group_objectives

	activation_description = {
		custom_tooltip = je_specialized_inputs_activation_tt
	}

	immediate = {
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				any_scope_building = {
					filter = {
						OR = {
							is_building_type = building_rye_farm
							is_building_type = building_wheat_farm
							is_building_type = building_maize_farm
							is_building_type = building_millet_farm
						}
					}
					percent >= 0.75
					OR = {
						has_active_production_method = pm_tractors
						has_active_production_method = pm_compression_ignition_tractors
					}
					weekly_profit > 0
					occupancy >= 0.5
					is_subsidized = no
				}
				any_scope_building = {
					filter = {
						OR = {
							is_building_type = building_coal_mine
							is_building_type = building_iron_mine
							is_building_type = building_lead_mine
							is_building_type = building_sulfur_mine
						}
					}
					percent >= 0.75
					OR = {
						has_active_production_method = pm_steam_donkey_building_coal_mine
						has_active_production_method = pm_steam_donkey_mine
					}
					weekly_profit > 0
					occupancy >= 0.5
					is_subsidized = no
				}
				any_scope_building = {
					is_building_type = building_railway
					or = {
						level >= 5
					}
				}
			}
		}
		effect = {
			set_variable = player_obj_home_focus
			je_specialized_inputs_cleanup = yes
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			has_variable = player_obj_export_focus
		}
		effect = {
			je_specialized_inputs_cleanup = yes
		}
	}

	weight = 1002
	transferable = yes
}
