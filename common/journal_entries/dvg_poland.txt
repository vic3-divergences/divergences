je_pol_early_industrial_development = {

	#You should build factories! It's fun for everyone!	
	
	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	group = je_group_technology

	is_shown_when_inactive = {
		OR = {
			AND = {
				exists = c:POL
				c:POL = ROOT
			}
			AND = {
				exists = c:PLC
				c:PLC = ROOT
			}
		}
	}
	
	possible = {
		has_technology_researched = atmospheric_engine
		has_technology_researched = railways
		owns_entire_state_region = STATE_GREATER_POLAND
		OR = {
			#Ough i wish you could use every scope state as a trigger - dont put underscores in here because you'll pollute the search!
			any_scope_state = {
				is_in_geographic_region = geographic_region_lesser_poland
				owner = ROOT
				count >= 4
			}
			hidden_condition = {
				is_in_civil_war = yes
			}
		}
	}
	
	complete = {
		custom_tooltip = {
			text = dvg_pol_vistula_works_complete_yes_tt
			has_variable = dvg_pol_vistula_works_complete_yes_var
		}
	}
	
	on_complete = {
		trigger_event = poland.100
	}
	
	fail = {
		OR = {
			any_scope_state = {
				count >= 4
				AND = {	
					is_in_geographic_region = geographic_region_lesser_poland
					NOT = {
						owner = ROOT
					}
				}
			}
			AND = {
				NOT = {
					owns_entire_state_region = STATE_GREATER_POLAND
				}
				hidden_condition = {
					NOT = {
						is_in_civil_war = yes
					}
				}
			}
		}
	}
	
	on_fail = {
		trigger_event = poland.101
	}
	
	#Main buttons
	scripted_button = dvg_pol_vistula_finish_works_button
	scripted_button = dvg_pol_vistula_mouth_button
	scripted_button = dvg_pol_vistula_lublin_canal_button
	scripted_button = dvg_pol_vistula_connecting_krakow_button
	
	#Bonus buttons
	scripted_button = dvg_pol_vistula_warta_works_button
	scripted_button = dvg_pol_vistula_krakow_portage_button
	scripted_button = dvg_pol_vistula_connecting_eastern_poland_button
}
