    icon = "gfx/interface/icons/event_icons/event_broken_crown.dds"

	group = je_group_internal_affairs

    complete = {
        OR = {
            AND = {
                ig:ig_landowners = {
			        is_powerful = yes
			        is_in_government = yes
		        }
		        has_insurrectionary_interest_groups = no
                owns_entire_state_region = STATE_MAZOVIA
                owns_entire_state_region = STATE_GREATER_POLAND
                has_law = law_type:law_hereditary_bureaucrats
                
                has_law = law_type:law_serfdom
                has_law = law_type:law_oligarchy
            }
            AND = {
                ig:ig_landowners = {
			        is_powerful = no
			        is_in_government = no
		        }
                has_insurrectionary_interest_groups = no
                owns_entire_state_region = STATE_MAZOVIA
                owns_entire_state_region = STATE_GREATER_POLAND
                has_law = law_type:law_interventionism
                has_law = law_type:law_wealth_voting
                NOT = {
                    has_law = law_type:law_hereditary_bureaucrats
                    has_law = law_type:law_serfdom
					has_law = law_type:law_agrarianism
                }	
            }
        }
    }

    on_complete = {
        if = {
            limit = {
                ig:ig_landowners = {
			        is_powerful = yes
			        is_in_government = yes
		        }
            }
            trigger_event = {
			    id = poland.2
		    }
        }
        else = {
            trigger_event = {
			    id = poland.3
		    }
        }
    }

    timeout = 16425

    on_timeout = {
        if = {
            limit = {
                ig:ig_landowners = {
			        is_powerful = yes
			        is_in_government = yes
		        }
            }
            trigger_event = {
			    id = poland.2
		    }
        }
        else = {
            trigger_event = {
			    id = poland.3
		    }
        }
	}

    weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
	
	scripted_button = dvg_pol_transfer_plock_button
}

			text = pol_maximilian_died
			c:POL = {
				ruler = {
					NOT = { has_variable = is_maximilian_jagiello }
				}
			}
		}
	}

	on_complete = {
		custom_tooltip = pol_east_withdraws
		if = {
			limit = {
				exists = c:POL
				c:POL = ROOT
			}
			trigger_event = { id = poland.10 }
		}
		if = {
			limit = {
				exists = c:POL
				c:POL = ROOT
			}
			remove_modifier = diplomatic_plays_blocked
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					exists = c:POL
					c:POL = ROOT
					ruler = { has_variable = is_maximilian_jagiello }
					year >= 1840
				}
				ruler = { kill_character = yes }
			}
			if = {
				limit = {
					exists = c:POL
					c:POL = ROOT
					ruler = { has_variable = is_maximilian_jagiello }
					year >= 1839
				}
				random_list = {
					8 = { }
					4 = {
						ruler = { kill_character = yes }
					}
				}
			}
		}
	}

	invalid = {
		OR = {
			NOT = {
				exists = c:POL
			}
		}
	}

	weight = 10

	should_be_pinned_by_default_uninvolved_or_context = yes
}
