je_cossack_east_expansion = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	group = je_group_foreign_affairs

	is_shown_when_inactive = {
		OR = {
			AND = {
					exists = c:ORE
        		THIS = c:ORE
			}
			AND = {
				country_has_primary_culture = cu:cossack
				scope:capital_scope = {
					state_region = {
						NOT = {
							is_homeland = cu:cossack
						}
					}
				}
			}
		}
      
       
    }

	possible = {
		has_technology_researched = nationalism
        
    }


	immediate = {
		
	}

	complete = {
		is_country_type = recognized
		OR = {
			AND = {
			exists = c:URC
			c:URC = this
			}
			AND = {
			exists = c:CSK
			c:CSK = this
			}
		}
		
	}

    on_complete = {
		every_scope_state = {
			limit = {
				OR = {
					is_in_geographic_region = geographic_region_siberia
					is_in_geographic_region = geographic_region_russia
					is_in_geographic_region = geographic_region_central_asia
					is_in_geographic_region = geographic_region_caucasus
				}
				NOT = {
					state_region = {
						is_homeland = cu:cossack
					}
				}
			}
			state_region = {
				add_homeland = cu:cossack	
			}
		}
	}
	
	weight = 1000
	should_be_pinned_by_default_uninvolved_or_context = yes
}
