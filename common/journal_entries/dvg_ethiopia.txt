        conclusion_desc = je_standard_completion_conclusion_desc

			has_law = law_type:law_freedom_of_conscience
		}

		}

    }

        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = je_standard_failure_conclusion_desc

			scope:journal_entry = {

			}
		}

			}
			hidden_effect = {

					activate_law = law_type:law_theocracy

						historical = yes
						first_name = Mehmet
						last_name = al_Mahdi
						ruler = yes
						is_general = yes

						interest_group = ig_devout
						ig_leader = yes
						ideology = ideology_theocrat
						traits = {
							arrogant
							imperious
							popular_commander
						}
					}
				}
				create_diplomatic_play = {
					target_country = c:SUD
					
					war = yes
					
					type = dp_annex_war
					
					add_war_goal = {
						holder = c:SUD
						type = humiliation
						target_country = c:ETH
					}
				}
			}			
		}
    }	

    outcome = {
        flags = { hidden invalid_outcome }
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {
			NOT = { owns_entire_state_region = STATE_BLUE_NILE }
		}
        effect = {}
    }

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					any_scope_state = {
						radical_fraction = { 
							value >= 0.25
							culture = cu:sudanese
						}
					}
				}
				change_variable = {
					name = mahdi_var
					add = 10
				}
			}
			if = {
				limit = {
					any_scope_state = {
						radical_fraction = { 
							value >= 0.50
							culture = cu:sudanese
						}
					}
				}
				change_variable = {
					name = mahdi_var
					add = 10
				}
			}

			clamp_variable = { name = mahdi_var max = 100 min = 0 }
		}
	}

	current_value = {
		value = root.var:mahdi_var
	}

	goal_add_value = {
		value = 100
	}
	
	progressbar = yes
	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
}
