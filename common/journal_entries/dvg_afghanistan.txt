je_panjabi_revolt = {
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	group = je_group_internal_affairs

	is_shown_when_inactive = {
		
	}

	possible = {
		
	}

	immediate = {
		set_variable = {
			name = punjab_var
			value = 0
		}
	}

	on_monthly_pulse = {
		random_events = {
			80 = 0
			10 = dvg_afghanistan.5
			10 = dvg_afghanistan.6
		}
		effect = {
			
			if = {
				limit = {
					any_scope_pop = {
						standard_of_living < 6.0 
						religion = rel:sunni
					}
				}
				every_scope_pop = {
					limit = {
						standard_of_living < 6.0
						religion = rel:sunni
					}
					change_pop_religion = {
						target = rel:sikh
						value = 0.2
					}
				}
			}  
		}
	}

	complete = {
		calc_true_if = {
			amount > 2 

			has_technology_researched = nationalism

			country_average_culture_and_religion_pop_acceptance = {
				target = cu:panjabi
				religion = rel:sikh
				value < 60
			}
			any_scope_state = {
				state_region = {
						is_homeland = cu:panjabi
					}
				radical_fraction = {
					value > 0.25
				}
			}
			cu:panjabi = {
				culture_current_fervor > 29
			}
		}
	}
	
	on_complete = {
		#set_variable = dvg_panjabi_revolt_complete
		trigger_event = {
			id = dvg_afghanistan.2
		}
	}
	
	fail = {
		has_technology_researched = pan-nationalism
		calc_true_if = {
			amount > 2


			country_average_culture_and_religion_pop_acceptance = {
				target = cu:panjabi
				religion = rel:sikh
				value > 60
			}

			cu:panjabi = {
				culture_current_fervor < 20
			}
			average_sol_for_religion = {
				target = rel:sikh
				value > 14.9
			}
			average_sol_for_culture = {
				target = cu:panjabi
				value > 14.9
			}
		}
		
	}
	
	on_fail = {
		set_variable = dvg_sikh_accepted
		trigger_event = {
			id = dvg_afghanistan.3
		}
	}

	invalid = { 
	NAND = {
			has_state_in_state_region = STATE_PUNJAB
			has_state_in_state_region = STATE_HILL_PUNJAB
		}
	}
	
	current_value = {
		value = root.var:punjab_var
	}

	goal_add_value = {
		value = 300
	}

	progressbar = yes

	weight = 10000
	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_panjabi_revolt2 = {
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	group = je_group_internal_affairs

	is_shown_when_inactive = {
		any_scope_state = {
			state_region = s:STATE_PUNJAB
			
		}
	}

	possible = {
		cultural_acceptance_base = { target = cu:panjabi value < acceptance_status_5 }
	}

	immediate = {
		set_variable = {
			name = punjab_var
			value = 0
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					cultural_acceptance_base = { target = cu:panjabi value < acceptance_status_5 }
				}
				change_variable = {
					name = punjab_var
					add = 1
				}
			}
		}
	}

	complete = {
		# If-else clause to avoid errors because of missing variable before JE activates
		trigger_if = {
			limit = { exists = var:punjab_var }
			scope:journal_entry = { is_goal_complete = yes }
		}
		is_at_war = yes
		
		
		
	}
	
	on_complete = {
		#set_variable = dvg_panjabi_revolt_complete
		trigger_event = {
			id = dvg_persia.2
		}
	}
	
	fail = {
		NOT = { has_law = law_type:law_total_separation }
		OR = {
			NOT = {owns_entire_state_region = STATE_PUNJAB}
			cultural_acceptance_base = { target = cu:panjabi value >= acceptance_status_5 }
		}
		

	}
	
	on_fail = {
		set_variable = dvg_sikh_accepted
		trigger_event = {
			id = dvg_persia.3
		}
	}

	
	current_value = {
		value = root.var:punjab_var
	}

	goal_add_value = {
		value = 48
	}

	progressbar = yes

	weight = 10000
	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_afghan_nation = {
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	group = je_group_internal_affairs

	is_shown_when_inactive = {
		exists = c:AFG
		c:AFG = this
	}

	possible = {
		has_technology_researched = nationalism
	}

	immediate = {
		
	}

	on_monthly_pulse = {
		fallback = on_yearly_events

	}

	complete = {
		calc_true_if = {
			amount > 3 

			has_technology_researched = pan-nationalism

			NOT = {
				has_law = law_type:law_subjecthood
			}
			is_country_type = recognized
			AND = {
				owns_entire_state_region = STATE_PASHTUNISTAN
				owns_entire_state_region = STATE_HERAT
				owns_entire_state_region = STATE_CENTRAL_HIGHLANDS
				owns_entire_state_region = STATE_KABUL
			}
			legitimacy > 75

			AND = {
				exists = c:PER
				c:PER = this
			}
		}
	}
	
	on_complete = {
		#set_variable = dvg_panjabi_revolt_complete
		trigger_event = {
			id = dvg_afghanistan.2
		}
	}
	
	fail = {
		calc_true_if = {
			amount > 2

			has_variable = recent_capitulation

			AND = {
				radical_fraction = {
				culture = cu:panjabi
				value > 0.33
				}
				any_scope_state = {
					state_region = {
						is_homeland = cu:panjabi
					}
				}
				cu:panjabi = {
				culture_current_fervor > 50
				}
			}
			AND = {
				radical_fraction = {
				culture = cu:pashtun
				value > 0.33
				}
				any_scope_state = {
					state_region = {
						is_homeland = cu:pashtun
					}
					radical_fraction = {
						value > 0.33
					}
				}
				cu:pashtun = {
				culture_current_fervor > 50
				}
			}
			AND = {
				radical_fraction = {
				culture = cu:hazara
				value > 0.33
				}
				any_scope_state = {
					state_region = {
						is_homeland = cu:hazara
					}
					radical_fraction = {
						value > 0.33
					}
				}
				cu:hazara = {
				culture_current_fervor > 50
				}
			}
			AND = {
				radical_fraction = {
				culture = cu:tajik
				value > 0.33
				}
				any_scope_state = {
					state_region = {
						is_homeland = cu:tajik
					}
					radical_fraction = {
						value > 0.33
					}
				}
				cu:tajik = {

		}
	}
	
	current_value = {
