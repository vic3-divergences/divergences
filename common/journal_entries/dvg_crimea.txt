je_crimea_succession = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	group = je_group_internal_affairs

	is_shown_when_inactive = {
		exists = c:CRI
		ROOT = c:CRI
		ruler = {
			has_variable = is_old_selim_giray
		}
	}

    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = je_standard_completion_conclusion_desc
        trigger = {
			custom_tooltip = {
				text = je_crimea_succession_complete_tt
				ruler = {
					NOT = { has_variable = is_old_selim_giray }
				}
			}
		}
        effect = {
			if = {
				limit = {
					ruler = {
						has_variable = is_murad_giray
					}
				}
				trigger_event = { id = dvg_kipchak.3 }
			}
		}
        event_outcome_desc = {} # complex desc like a status_desc field
    }

	on_monthly_pulse = {
		effect = {
			random_list = {
				9 = { }
				3 = {
					trigger_event = { id = dvg_kipchak.2 }
				}
			}
			if = {
				limit = {
					has_variable = murad
				}
				random_list = {
					2 = { }
					1 = {
						ruler = { kill_character = yes }
					}
				}
			}
			if = {
				limit = {
					year >= 1838
				}
				ruler = { kill_character = yes }
			}
		}
	}

	weight = 1000

	should_be_pinned_by_default_uninvolved_or_context = yes
}

	group = je_group_internal_affairs

    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = je_standard_completion_conclusion_desc
        trigger = {
			army_size >= 50
			has_technology_researched = napoleonic_warfare
			NOR = {
				any_scope_building = {
					is_building_type = building_barrack

						building_has_goods_shortage = yes
						has_active_production_method = pm_no_organization
					}
				}
				any_military_formation = {
					filter = {
						is_army = yes
					}
					any_combat_unit = {
						has_unit_type = unit_type:combat_unit_type_irregular_infantry
					}
					percent >= 0.25
				}
			}			
		}
        effect = {
			add_modifier = {
				name = murad_army_reform
				days = normal_modifier_time
			}			
		}
        event_outcome_desc = {} # complex desc like a status_desc field
    }

    outcome = {
        flags = { hidden invalid_outcome }
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {
			ruler = {
				NOT = { has_variable = is_murad_giray }
			}			
		}
        effect = {}
    }

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					is_ai = yes
					has_law = law_type:law_peasant_levies
				}
				activate_law = law_type:law_peasant_levies
			}
		}
	}

	transferable = no

	weight = 5000
}
