je_dvg_greek_nationalism = {
	icon = "gfx/interface/icons/invention_icons/nationalism.dds"
	
	group = je_group_historical_content

	is_shown_when_inactive = {
		any_scope_state = {
			state_region = {
				is_homeland = cu:greek
			}
		}
		#exists = c:VEN
		#c:VEN = ROOT
		any_scope_character = {
			has_variable = zimisces_var
		}
		
		NAND = {
			exists = c:GRE
			c:GRE = ROOT
		}
	}

	possible = {
		OR = {
			has_technology_researched = nationalism
			AND = {
				exists = c:GRE
				c:GRE = {
					has_technology_researched = nationalism
				}
			}
			any_enemy_in_diplo_play = {
				country_has_primary_culture = cu:greek
			}
		}
	}

	immediate = {
		
		if = {
			limit = {
				any_scope_character = {
					has_variable = zimisces_var
				}
				exists = c:VEN
				c:VEN = ROOT
				
			}
			trigger_event = {
				id = dvg_greece.201
			}

		}
		set_variable = {
			name = greece_var
			value = 0
		}
		if = {
			limit = {
				NOT = {
					has_technology_researched = nationalism
				}
			}
			add_technology_researched = nationalism
		}
	}
	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
					exists = c:GRE
					
					c:GRE = {
						ruler = {
							NOT = {
								has_variable = zimisces_var
							}
						}
						any_scope_character = {
							has_variable = zimisces_var
						}
					}
				}
				
			}
		}
		
	}
	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					cultural_acceptance_base = { target = cu:greek value < acceptance_status_4 }	
					exists = c:GRE
				}
				c:GRE = {
					add_liberty_desire = 0.10
				}
			}
			if = {
				limit = {
					cultural_acceptance_base = { target = cu:greek value >= acceptance_status_4 }
					exists = c:GRE
				}
				c:GRE = {
					add_liberty_desire = -0.20
				}
			}
			if = {
				limit = {
					country_has_voting_franchise = yes
					
					exists = c:GRE
				}
				c:GRE = {
					add_liberty_desire = -0.20
				}
			}
			
			if = {
				limit = {
					NOT = {
						exists = c:GRE
						has_variable = greece_to_liberate
					}
				}
				set_variable = greece_to_liberate

				
			}
			if = {
				limit = {
					has_variable = council_start
					NOT = {
						has_variable = greece_citizen_var
					}
				}
				trigger_event = {
					id = dvg_greece.210
				}
				
			}
			if = {
				limit = {
					has_variable = council_start
					NOT = {
						has_variable = greece_voting_var
					}
				}
				trigger_event = {
					id = dvg_greece.211
				}
			}
			if = {
			#Greek Integration

				limit = {
					has_variable = council_start
					has_variable= greece_citizen_var
					OR = {
						has_variable = greece_voting_var
						has_law = law_type:law_census_voting
					}	
					OR = {
						has_variable= greece_citizen_var
						cultural_acceptance_base = { target = cu:greek value >= acceptance_status_4 }
					}
					
				}
				trigger_event = {
					id = dvg_greece.212
				}
			}
			#Greek Revolt
			
		}
	}
	

	outcome = { #win
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			c:GRE = {
				liberty_desire > 90
			}
			has_variable = council_start
			has_variable = greece_voting_var
			has_variable= greece_citizen_var
			
		}
		effect = {
			if = {
				limit = {
					exists = c:GRE
					has_state_in_state_region = STATE_CYPRUS
				}
				trigger_event = {
					id = dvg_greece.203
				}
			}
		}
	}

	outcome = { #lose
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {
				cultural_acceptance_base = { target = cu:greek value >= acceptance_status_4 }
				
				has_law = law_type:law_oligarchy
				has_law = law_type:law_autocracy
			}
			c:GRE = {
				liberty_desire < 25
			}

		}
		effect = {
			trigger_event = {
				id = dvg_greece.212
			}
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			trigger_event = {
				id = dvg_greece.203
			}
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				has_variable = council_cancelled
				AND = {
					exists = c:GRE
					is_diplomatic_play_enemy_of = c:GRE
				}
				NOT = {
					any_scope_state = {
						state_region = {
							is_homeland = cu:greek
						}
					}
					any_subject_or_below = {
						country_has_primary_culture = cu:greek
					}
				}
			}
			
			
		}
		effect = {
		}
	}

	weight = 500
	should_be_pinned_by_default_uninvolved_or_context = yes
	goal_add_value = {
		value = 48
	}
	timeout = 180
	
}

je_aragocracy = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	is_shown_in_lobby = {
		has_dlc_feature = ip3_content #National Awakening
		c:MCD ?= THIS
	}

	scripted_button = je_bavarocracy_convert_button

	scripted_progress_bar = bavarocracy_progress_bar

	should_be_pinned_by_default_uninvolved_or_context = yes
	transferable = no
	weight = 10000

	modifiers_while_active = {
		modifier_foreign_administration
	}

	immediate = {
		random_scope_character = {
			limit = {
				has_template = GRE_otto_von_wittelsbach
			}
			save_scope_as = king_otto_scope
		}
		set_variable = {
			name = je_bavarocracy_timer_var
			value = 0
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					scope:journal_entry = {
						"scripted_bar_progress(bavarocracy_progress_bar)" >= 80
					}
				}
				change_variable = {
					name = je_bavarocracy_timer_var
					add = 1
				}
			}
		}
	}

	outcome = { #win
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				AND = {
					has_law_or_variant = law_type:law_autocracy
					institution_investment_level = {
						institution = institution_home_affairs
						value >= 1
					}
				}
				AND = {
					country_has_voting_franchise = yes
					has_law_or_variant = law_type:law_elected_bureaucrats
					OR = {
						has_law_or_variant = law_type:law_right_of_assembly
						has_law_or_variant = law_type:law_protected_speech
					}
				}
			}
			government_legitimacy >= 75
			custom_tooltip = {
				text = je_bavarocracy_complete_tt
				scope:journal_entry = {
					"scripted_bar_progress(bavarocracy_progress_bar)" < 10
				}
			}
		}
		effect = {
			trigger_event = {
				id = dvg_macedonia.1
				popup = yes
			}
			hidden_effect = {
				scope:king_otto_scope ?= {
					if = {
						limit = {
							is_immortal = yes
						}
						set_character_immortal = no
					}
				}
			}
		}
	}

	event_outcome_completed_desc = {
		first_valid = {
			triggered_desc = {
				desc = event_effects_dvg_macedonia.1_tt
			}
		}
	}

	outcome = { #lose
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = je_bavarocracy_fail_tt
				var:je_bavarocracy_timer_var >= 50
			}
		}
		effect = {
			trigger_event = {
				id = dvg_macedonia.2
				popup = yes
			}
			hidden_effect = {
				scope:king_otto_scope ?= {
					if = {
						limit = {
							is_immortal = yes
						}
						set_character_immortal = no
					}
				}
			}
			custom_tooltip = bavarocracy_fail_tt
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				NOT = {
					has_law = law_type:law_monarchy
				}
				c:GRE ?= {
					has_variable = ruler_selector_var
				}
			}
		}
		effect = {
			scope:king_otto_scope ?= {
				if = {
					limit = {
						is_immortal = yes
					}
					set_character_immortal = no
				}
			}
		}
	}
}
