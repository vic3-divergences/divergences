law_peasant_levies = {
	group = lawgroup_army_model
	icon = "gfx/interface/icons/law_icons/peasant_levies.dds"

	progressiveness = -50

	modifier = {
		country_can_only_conscript_peasants_bool = yes
		unit_morale_loss_mult = 0.1
		unit_experience_gain_mult = -0.25
		state_conscription_rate_add = 0.04
		country_aristocrats_pol_str_mult = 0.25
		country_military_goods_cost_mult = -0.1
		state_building_barrack_max_level_add = 25
		state_building_conscription_center_max_level_add = 25
	}

	can_impose = {
		OR = {
			AND = {
				is_in_same_power_bloc = scope:target_country
				AND = {
					modifier:country_can_impose_same_lawgroup_army_model_in_power_bloc_bool = yes
					has_law = scope:law
				}
			}
			can_impose_law_default = yes
		}
	}

	is_visible = {
		NOR = {
			country_has_primary_culture = cu:japanese
			country_has_primary_culture = cu:kaiganese
		}
	}

	on_enact = {
		custom_tooltip = QUALIFICATIONS_OFFICERS_PEASANT_LEVIES_DESC
		if = {
			limit = {
				NOT = {
					has_law_or_variant = law_type:law_peasant_levies
				}
			}
			custom_tooltip = aristocrats_af_ig_shift_effect_tt
		}
		if = {
			limit = {
				has_law_or_variant = law_type:law_professional_army
			}
			custom_tooltip = military_pops_af_ig_shift_effect_negative_tt
			custom_tooltip = non_military_pops_af_ig_shift_effect_negative_tt
		}
	}

	# AI will never return to peasant levies
	ai_will_do = {
		always = no
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}

law_professional_army = {
	group = lawgroup_army_model
	icon = "gfx/interface/icons/law_icons/professional_army.dds"

	progressiveness = 0

	unlocking_technologies = { military_drill }

	modifier = {
		unit_morale_loss_mult = -0.1
		state_conscription_rate_add = 0.01
		country_officers_pol_str_mult = 0.5
		state_building_barrack_max_level_add = 100
		state_building_conscription_center_max_level_add = 50
		unit_experience_gain_mult = 1.0
	}

	can_impose = {
		OR = {
			AND = {
				is_in_same_power_bloc = scope:target_country
				AND = {
					modifier:country_can_impose_same_lawgroup_army_model_in_power_bloc_bool = yes
					has_law = scope:law
				}
			}
			can_impose_law_default = yes
		}
	}

	disallowing_laws = {
		law_strict_edo_system
		law_intermediate_edo_system
	}

	on_enact = {
		if = {
			limit = {
				has_law_or_variant = law_type:law_peasant_levies
			}
			custom_tooltip = aristocrats_af_ig_shift_effect_negative_tt
		}
		if = {
			limit = {
				NOT = {
					has_law_or_variant = law_type:law_professional_army
				}
			}
			custom_tooltip = non_military_pops_af_ig_shift_effect_tt
			custom_tooltip = military_pops_af_ig_shift_effect_tt
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}

		if = {
			limit = {
				has_law_or_variant = law_type:law_peasant_levies
			}
			add = 1000
		}

		# Once the AI gets to a certain army tech level they should try to ditch Peasant Levies
		if = {
			limit = {
				OR = {
					has_technology_researched = logistics
					has_technology_researched = general_staff
				}
				has_law_or_variant = law_type:law_peasant_levies
			}

			add = 1000
		}
	}
}

law_revolutionary_guard = {
	group = lawgroup_army_model
	icon = "gfx/interface/icons/law_icons/revolutionary_guard.dds"

	progressiveness = 100

	unlocking_technologies = { enlistment_offices }

	unlocking_laws = {
		law_council_republic
		law_factory_councils
	}

	modifier = {
		state_conscription_rate_add = 0.05
		building_training_rate_mult = 1
		country_leverage_resistance_mult = 0.5
		country_soldiers_pol_str_mult = 0.25
		country_officers_pol_str_mult = -0.5
		state_building_barrack_max_level_add = 25
		state_building_conscription_center_max_level_add = 100
	}

	on_activate = {
		custom_tooltip = non_revolutionary_generals_will_resign_tt
		custom_tooltip = new_revolutionary_generals_tt
		hidden_effect = {
			every_scope_character = {
				limit = {
					OR = {
						has_role_of_type = general
						has_role_of_type = admiral
					}
					NOT = {
						law_stance = {
							law = law_type:law_council_republic
							value > neutral
						}
					}
				}
				commander_military_formation = {
					save_scope_as = formation_scope
				}
				random_list = {
					1 = {
						modifier = {
							add = average_progressiveness
						}
						# Remains in office
					}
					75 = {
						unassign_from_formation = yes
						if = {
							limit = {
								has_role_of_type = general
							}
							owner = {
								create_character = {
									interest_group = ig_trade_unions
									is_general = yes
									trait_generation = {
										add_random_trait = personality
										add_random_trait = skill
									}
									on_created = {
										assign_random_revolutionary_ideology = yes
										if = {
											limit = {
												exists = scope:formation_scope
											}
											transfer_to_formation = scope:formation_scope
										}
									}
								}
							}
						}
						else_if = {
							limit = {
								has_role_of_type = admiral
							}
							owner = {
								create_character = {
									interest_group = ig_trade_unions
									is_admiral = yes
									trait_generation = {
										add_random_trait = personality
										add_random_trait = skill
									}
									on_created = {
										assign_random_revolutionary_ideology = yes
										if = {
											limit = {
												exists = scope:formation_scope
											}
											transfer_to_formation = scope:formation_scope
										}
									}
								}
							}
						}
						retire_character = yes
					}
				}
			}
		}
	}

	on_enact = {
		if = {
			limit = {
				has_law_or_variant = law_type:law_peasant_levies
			}
			custom_tooltip = aristocrats_af_ig_shift_effect_negative_tt
		}
		if = {
			limit = {
				has_law_or_variant = law_type:law_professional_army
			}
			custom_tooltip = military_pops_af_ig_shift_effect_negative_tt
			custom_tooltip = non_military_pops_af_ig_shift_effect_negative_tt
		}
	}

	ai_will_do = {
		NOT = { has_strategy = ai_strategy_consolidation_agenda }
		any_interest_group = {
			is_in_government = yes
			has_party = yes
			party ?= {
				OR = {
					is_party_type = socialist_party
					is_party_type = communist_party
					is_party_type = anarchist_party
				}
			}
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}
