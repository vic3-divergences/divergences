law_incan_economics = {
	group = lawgroup_economic_system
	parent = law_command_economy # Inherits ideological stances from parent

	icon = "gfx/interface/icons/law_icons/manorialism.dds"

	progressiveness = 100

	is_visible = {
		OR = {
			c:TWT ?= this # For the Incan Empire and its subjects
			AND = {
				exists = c:TWT
				is_subject_of = c:TWT
				OR = {
					is_subject_type = subject_type_crown_land
					is_subject_type = subject_type_personal_union
				}
				c:TWT ?= { has_law_or_variant = law_type:law_serfdom }
			}
			country_has_primary_culture = cu:quechua
			country_has_primary_culture = cu:aimara
		}
	}

	on_activate = {
		seize_investment_pool = yes
	}

	modifier = {
		country_disable_investment_pool_bool = yes

		country_authority_mult = 0.25
		country_bureaucrats_pol_str_mult = 0.25
		interest_group_ig_landowners_pol_str_mult = 0.25
		interest_group_ig_armed_forces_pol_str_mult = 0.25
		building_nationalization_cost_mult = -0.5
		building_nationalization_radicals_mult = -0.75
		country_government_dividends_efficiency_add = 0.4
		country_private_construction_allocation_mult = 0.1
		building_company_government_dividends_add = 0.75
		state_tax_capacity_mult = -0.25
		country_free_charters_add = 1

		country_leverage_resistance_mult = 0.25

		country_disable_non_company_privatization_bool = yes
	}

	ai_will_do = {
		OR = {
			has_law = law_type:law_technocracy
			AND = {
				has_law = law_type:law_council_republic
				ruler ?= {
					has_ideology = ideology:ideology_vanguardist
					has_ideology = ideology:ideology_communist
					
				}
			}
			any_interest_group = {
				has_ideology = ideology:ideology_inka_paternalistic
			}
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}

		if = {
			limit = {
				has_law_or_variant = law_type:law_council_republic
			}
			add = 250
		}

		if = {
			limit = {
				has_law_or_variant = law_type:law_technocracy
			}
			add = 100
		}
	}

	ai_impose_chance = {
		value = 0

		if = {
			limit = {
				has_law_or_variant = law_type:law_command_economy
			}

			add = base_impose_law_weight
			multiply = 10.0
		}
	}
}
