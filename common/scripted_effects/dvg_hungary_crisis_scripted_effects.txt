dvg_hungary_crisis_boh_power_gain = {
	custom_tooltip  = DVG_HUNGARIAN_CRISIS_BOH_POWER_GAIN

	every_country = {
		limit = {
			has_variable = dvg_attend_hungarian_conference
		}
		change_variable = {
			name = dvg_hungarian_crisis_boh_power
			add = 1
		}
	}
	
}
