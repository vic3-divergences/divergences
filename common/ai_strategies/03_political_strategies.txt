		}

		if = {
			limit = {
				has_enactment_je_or_law_commitment = yes

	type = political

	anti_interest_groups = {

	pro_movements = {

		movement_religious_majority
		movement_cultural_majority
		movement_reactionary
		movement_pro_slavery
	}

	anti_movements = {
		movement_socialist
		movement_labor
		movement_communist
		movement_anarchist
		movement_nihilist
		movement_radical
		movement_liberal
		movement_german_national
		movement_italian_national
		movement_modernizer
		movement_feminist
		movement_positivist
		movement_cultural_minority
		movement_religious_minority
		movement_land_reform
	}


	institution_scores = {
		institution_colonial_affairs = {
			value = 100
		}
		institution_social_security = {
			value = 50
		}
		institution_workplace_safety = {
			value = 50
		}
		institution_schools = {
			value = 75
		}
		institution_police = {
			value = 200
		}
		institution_health_system = {
			value = 75
		}
		institution_home_affairs = {
			value = 200
		}
	}

	possible = {
		OR = {
			AND = {
				OR = {
					AND = {
						exists = ig:ig_landowners
						ig:ig_landowners ?= { ig_counts_as_marginal = no }
					}
					AND = {
						exists = ig:ig_devout
						ig:ig_devout ?= { ig_counts_as_marginal = no }
					}
				}
				AND = { # need something to return to first
					has_technology_researched = egalitarianism
					country_has_laws_in_need_of_critical_modernization = yes
				}
			}
		}
	}

	weight = {
		value = 5

		if = {
			limit = {
				ig:ig_landowners ?= { is_powerful = yes }
			}

			add = 5
		}

		if = {
			limit = {
				ig:ig_devout ?= { is_powerful = yes }
			}

			add = 5
		}

		if = {
			limit = {
				ruler ?= {
					average_progressiveness <= 25
				}
			}
			add = 5

			if = {
				limit = {
					OR = {
						has_law_or_variant = law_type:law_monarchy
						has_law_or_variant = law_type:law_theocracy
						has_law_or_variant = law_type:law_autocracy
						has_law_or_variant = law_type:law_oligarchy
						has_law_or_variant = law_type:law_landed_voting
					}
				}
				add = 5
			}
		}

		if = {
			limit = {
				has_modifier = forced_regime_change
				OR = {
					ig:ig_devout ?= { is_in_government = yes }
					ig:ig_landowners ?= { is_in_government = yes }
				}
			}

			add = 1000
		}

		if = {
			limit = {
				has_radical_leftist_government = yes
			}
			multiply = {
				value = 0.75
				subtract = modifier:country_other_country_socialist_government_hostility_impact_mult
				min = 0
				max = 1
			}
		}

	}
}
