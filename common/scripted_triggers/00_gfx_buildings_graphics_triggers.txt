# Root = State Region
us_american_buildings_trigger = {
	is_in_geographic_region = geographic_region_north_america
	any_scope_state = {
		owner = {
			OR = {
				country_has_primary_culture = cu:yankee
				country_has_primary_culture = cu:dixie
				country_has_primary_culture = cu:afro_american
				country_has_primary_culture = cu:cajun
			}
		}
	}
}

latin_american_buildings_trigger = {

		OR = {
			region = sr:region_brazil
			region = sr:region_southern_europe
			region = sr:region_andes
			region = sr:region_la_plata
			region = sr:region_gran_colombia
			region = sr:region_mexico
			region = sr:region_caribbean
		}
	}
}

		is_in_geographic_region = geographic_region_france_old

			state_region = s:STATE_ALSACE
			state_region = s:STATE_LORRAINE

			state_region = s:STATE_CORSICA
			state_region = s:STATE_WEST_SWITZERLAND

		}
	}
}
